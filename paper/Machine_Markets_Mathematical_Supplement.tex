\section{Canonical notation and observation contracts}
In the retained discrete pressure-coordinate model,
\begin{align}
 x_{t+1}&=a x_t+\sum_{i=1}^{n}y_{i,t}+\kappa u_t+e_{t+1},\label{eq:disc-x}\\
 y_{i,t+1}&=(1-\eta_i)y_{i,t}+\eta_iq_i x_t,\label{eq:disc-y}
\end{align}
where $0\le a<1$, $q_i\ge0$, $0<\eta_i\le1$, and $\kappa>0$. Set
\[
 Q=\sum_iq_i,\qquad S=a+Q,\qquad G=\frac{Q}{1-a}.
\]
These are different quantities. In this model, $S<1$ is equivalent to Schur stability, but $S$ is not the numerical spectral radius. Nor is $G$ the closed-loop steady-state multiplier: for a stable system and a constant input, that multiplier is $\kappa/(1-a-Q)$.

With zero adaptive initial state, define
\begin{align}
 k_j&=\sum_i q_i\eta_i(1-\eta_i)^j,\label{eq:kernel}\\
 h_0&=1,\quad h_1=a,\quad h_{t+1}=a h_t+\sum_{j=0}^{t-1}k_jh_{t-1-j},\label{eq:recurrence-h}\\
 \beta_j&=\kappa h_j.\label{eq:beta}
\end{align}
The observable response is $\beta$, not the private state vector and not the parameter vector $(a,q)$.

\textbf{I0: controlled price responses.} The observer receives randomized probe magnitudes and episode price responses, together with an externally justified rate floor. Exact rates, private states, parameters, and covariance are not supplied.

\textbf{I1: audited architecture.} Exact prices, disclosed rates, verified resets, and a complete relevant channel inventory permit reconstructed regressors and the terminal-row pivot. This is stronger information than I0.

\textbf{I2: observational exchange records.} Public quotes and trades do not automatically provide exogenous interventions, independent repetitions, exact resets, or the maintained structural model. The theorems below apply to I2 only after the relevant observation and sampling assumptions are established.

\section{Continuous-time stability and the correct sampling limit}
\subsection{A positive-vector lemma}
A real matrix is \emph{Metzler} when every off-diagonal entry is nonnegative. It is \emph{Hurwitz} when all eigenvalues have negative real part. The following established criterion is included with a proof so the application is self-contained \cite{rantzer}.

\begin{lemma}[Positive-vector characterization]\label{lem:metzler}
For a finite-dimensional Metzler matrix $A$,
\[
 A\text{ is Hurwitz}\quad\Longleftrightarrow\quad
 \exists v>0\text{ such that }Av<0.
\]
All inequalities between vectors are componentwise.
\end{lemma}
\begin{proof}
Choose $c$ large enough that $A+cI\ge0$. Then
$e^{At}=e^{-ct}e^{(A+cI)t}\ge0$ for $t\ge0$, by the exponential series.
If $A$ is Hurwitz, define
\[
 v=\int_0^\infty e^{At}\one\,dt.
\]
The integral converges, and every coordinate is positive because the integrand equals $\one$ at zero and is continuous. Integrating its derivative yields $Av=-\one<0$.

Conversely, if $v>0$ and $Av<0$, let $D=\diag(v)$ and $\widetilde A=D^{-1}AD$. This matrix is Metzler, similar to $A$, and has strictly negative row sums. The rightmost real coordinate of its $i$th Gershgorin disk is its row sum. Every disk therefore lies in the open left half-plane, so every eigenvalue of $A$ has negative real part.
\end{proof}

\begin{lemma}[Nonnegative discrete counterpart]\label{lem:schur}
For $P\ge0$, $\rho(P)<1$ if and only if some $v>0$ satisfies $Pv<v$.
\end{lemma}
\begin{proof}
If $\rho(P)<1$, the Neumann series gives $v=(I-P)^{-1}\one>0$ and $Pv=v-\one<v$. Conversely, scaling by $D=\diag(v)$ gives a nonnegative matrix with all row sums strictly below one. Its induced infinity norm, hence its spectral radius, is below one.
\end{proof}

\subsection{Heterogeneous continuous rates}
Let
\[
 \Gamma=\diag(\gamma_1,\ldots,\gamma_d)>0,
 \qquad E=\diag(\lambda_1,\ldots,\lambda_n)>0,
\]
and let $B\in\R_+^{d\times n}$, $C\in\R_+^{n\times d}$, and $g\ge0$. Consider
\begin{equation}
 \dot x=-\Gamma x+gBm,\qquad \dot m=E(Cx-m),\qquad
 A_c=\begin{bmatrix}-\Gamma&gB\\EC&-E\end{bmatrix}.
 \label{eq:ct}
\end{equation}

\begin{theorem}[Continuous-time positive-feedback boundary]\label{thm:ct}
The system \eqref{eq:ct} is exponentially asymptotically stable if and only if
\begin{equation}
 \rho\!\left(\Gamma^{-1}gBC\right)<1.\label{eq:ct-threshold}
\end{equation}
In particular, for $\Gamma=\gamma I_d$,
\[
 A_c\text{ is Hurwitz}\quad\Longleftrightarrow\quad g\rho(BC)<\gamma.
\]
This condition is independent of the positive diagonal memory rates in $E$.
\end{theorem}
\begin{proof}
Let $K=\Gamma^{-1}gBC\ge0$. Suppose $\rho(K)<1$ and put $x=(I-K)^{-1}\one>0$. Then
\[
 (\Gamma-gBC)x=\Gamma\one>0.
\]
For sufficiently small $\varepsilon>0$, set $m=Cx+\varepsilon\one>0$. The two block components of $A_c(x,m)^\top$ are
\[
 -\Gamma x+gBm=-\Gamma\one+\varepsilon gB\one<0,
 \qquad ECx-Em=-\varepsilon E\one<0.
\]
Lemma~\ref{lem:metzler} proves that $A_c$ is Hurwitz.

Conversely, if $A_c$ is Hurwitz, the same lemma provides $x>0,m>0$ such that
$-\Gamma x+gBm<0$ and $ECx-Em<0$. Because $E$ is positive diagonal, $m>Cx$. Nonnegativity then gives
\[
 \Gamma x>gBm\ge gBCx,
 \qquad Kx<x.
\]
Lemma~\ref{lem:schur} implies $\rho(K)<1$.
\end{proof}

\begin{proposition}[Boundary and supercritical behavior]\label{prop:ctboundary}
If $\rho(\Gamma^{-1}gBC)=1$, $A_c$ has a zero eigenvalue. If that spectral radius exceeds one, $A_c$ has a strictly positive real eigenvalue.
\end{proposition}
\begin{proof}
At equality, Perron--Frobenius gives $x\ge0$, $x\ne0$, with $gBCx=\Gamma x$. Set $m=Cx$. Then $A_c(x,m)^\top=0$.

For the strict case define, for $s\ge0$,
\[
 K(s)=g(sI+\Gamma)^{-1}B(sI+E)^{-1}EC.
\]
Its spectral radius is continuous, exceeds one at $s=0$, and tends to zero as $s\to\infty$. Some $s_*>0$ therefore satisfies $\rho(K(s_*))=1$. Choose a corresponding nonzero nonnegative eigenvector $x$ and set $m=(s_*I+E)^{-1}ECx$. Substitution gives $A_c(x,m)^\top=s_*(x,m)^\top$.
\end{proof}

Equality does not imply exponential divergence: it has at least a nondecaying mode. Supercriticality supplies an exponentially growing mode, not a claim that every initial condition grows or that a financial price must crash. Linear ODE solutions remain finite at every finite time. The result depends on positivity and fixed finite-dimensional coefficients; signed feedback and switching require other arguments.

\subsection{Common-rate spectral abscissa}
\begin{proposition}[Closed expression in the common-rate case]\label{prop:sa}
For $\Gamma=\gamma I_d$, $E=\lambda I_n$, and $\nu=\rho(BC)$,
\begin{equation}
 \smax(A_c)=
 \frac{-(\gamma+\lambda)+\sqrt{(\gamma-\lambda)^2+4g\lambda\nu}}{2}.
 \label{eq:sa}
\end{equation}
Here $\smax(A_c)=\max\{\Re s:s\in\sigma(A_c)\}$.
\end{proposition}
\begin{proof}
Let the right side be $s_+$. If $g\nu>0$, select $v\ge0$ with $BCv=\nu v$. Then $Cv\ne0$ and
$w=\lambda Cv/(s_++\lambda)$ produces an eigenvector $(v,w)$ at eigenvalue $s_+$. For any eigenvalue $s$ other than $-\gamma,-\lambda$, block elimination gives an eigenvalue
\[
 \frac{(s+\gamma)(s+\lambda)}{g\lambda}
\]
of $BC$. If $\Re s>s_+$, then $\Re s>\max(-\gamma,-\lambda)$ and
\[
 |(s+\gamma)(s+\lambda)|\ge(\Re s+\gamma)(\Re s+\lambda)
 >(s_++\gamma)(s_++\lambda)=g\lambda\nu,
\]
a contradiction. The excluded eigenvalues have real part at most $s_+$. When $g=0$, block triangularity proves the formula. When $\nu=0$ and $g>0$, the same elimination restricts eigenvalues to $-\gamma,-\lambda$; both occur in the resulting characteristic polynomial, giving $s_+=-\min(\gamma,\lambda)$.
\end{proof}
The proof does not apply a real-coefficient quadratic test to potentially complex eigenvalues of $BC$.

\subsection{Euler scaling, convergence, and step-size restrictions}
For a physical step $h>0$, forward Euler gives $P_h=I+hA_c$. In the common-rate notation of the original paper this means
\begin{equation}
 a_h=1-h\gamma,\qquad g_h=hg,\qquad \eta_h=h\lambda.
 \label{eq:euler-scale}
\end{equation}
One must scale the coupling and memory-update parameters, not only rename $1-a$.

\begin{theorem}[Euler-consistent boundary and convergence]\label{thm:euler}
If
\[
 0<h\le\min\!\left\{\min_i\gamma_i^{-1},\min_j\lambda_j^{-1}\right\},
\]
then $P_h\ge0$ and
\[
 P_h\text{ is Schur}\quad\Longleftrightarrow\quad A_c\text{ is Hurwitz}.
\]
For common $\gamma,\lambda$, this is exactly
$g_h\rho(BC)<1-a_h$. In any induced matrix norm, for $kh\le T$,
\begin{equation}
 \norm{(I+hA_c)^k-e^{khA_c}}
 \le \frac12 Th\norm{A_c}^2e^{T\norm{A_c}}.
 \label{eq:euler-error}
\end{equation}
\end{theorem}
\begin{proof}
The step-size restriction makes every entry of $P_h$ nonnegative. If $A_cv<0$ for some $v>0$, then $P_hv<v$, so $P_h$ is Schur. Conversely, if $P_h$ is Schur, Lemma~\ref{lem:schur} gives $P_hv<v$, hence $A_cv<0$. Lemma~\ref{lem:metzler} completes the stability equivalence. Substitution of \eqref{eq:euler-scale} gives the common-rate inequality.

The exponential series bounds the one-step defect by
$\norm{e^{hA_c}-I-hA_c}\le\tfrac12h^2\norm{A_c}^2e^{h\norm{A_c}}$.
Use the telescoping identity for $P_h^k-(e^{hA_c})^k$, together with
$\norm{P_h^j}\le e^{jh\norm{A_c}}$. There are $k$ terms, each bounded by
$\tfrac12h^2\norm{A_c}^2e^{kh\norm{A_c}}$. Since $kh\le T$, \eqref{eq:euler-error} follows. It gives convergence at grid points; piecewise-linear interpolation adds an $O(h)$ term on every bounded time interval.
\end{proof}

\begin{remark}[Exact sampling is different]
The exact sampled transition is $e^{hA_c}$. Its eigenvalues are $e^{h s_i}$, so it is Schur exactly when $A_c$ is Hurwitz, for every $h>0$. It need not retain the original simple Euler block structure. Forward Euler outside its admissible step range can be unstable even for a stable ODE: $A_c=-I$, $h=3$ gives $I+hA_c=-2I$. Thus the correct claim is not unrestricted discretization invariance.
\end{remark}

\subsection{Continuous hidden-memory tails}
For $\dot y_i=-\lambda_i y_i+\lambda_iq_i x$, zero initial memory gives
\[
 y_i(t)=q_i\lambda_i\int_0^t e^{-\lambda_i(t-s)}x(s)\,ds.
\]
Hence the loop-open kernel is $K(t)=\sum_iq_i\lambda_i e^{-\lambda_i t}$ and
\[
 \int_0^\infty K(t)\,dt=Q,
 \qquad \int_T^\infty K(t)\,dt=\sum_iq_ie^{-\lambda_iT}
 \le Q_{\max}e^{-\lambda_0T}
\]
when $Q\le Q_{\max}$ and every $\lambda_i\ge\lambda_0>0$. These identities follow by direct integration. They extend the audited-tail interpretation, not the ability to verify a rate floor from prices.

\section{Corrected group identification, including unknown rates}
\subsection{Aggregate equivalence}
Under a common rate $\eta$, summing \eqref{eq:disc-y} gives
\[
 Y_{t+1}=(1-\eta)Y_t+\eta Qx_t,\qquad Y_t=\sum_i y_{i,t}.
\]
For the same initial aggregate state, probe sequence, and aggregate disturbance law, prices depend on $q$ only through $Q$. Equal $Q$ therefore implies equal aggregate price laws. Under linear realization theory \cite{ho1966,hautus1969}, the unobserved cohort decomposition lies in an unobservable eigenspace defect. Furthermore, separating distinct exponential rate mixtures incurs fundamental minimax information-theoretic rate limits \cite{chen1995optimal,heinrich2018,wuyang2020}. This is the retained symmetry result, not a statement about known heterogeneous rates.

\subsection{Known-rate measurement theorem}
Let experiment $j$ inject the channel-input vector $U_{j,:}$ at time zero, with known pulse units and otherwise zero inputs. From a zero state,
\[
 y_{i,1}=\eta q_iU_{j,i},\qquad x_1=0,\qquad x_2=\eta U_{j,:}q.
\]
Stacking the measured lag-2 responses yields $b=\eta Uq$.

\begin{theorem}[Uniform identification with a known common rate]\label{thm:group-known}
Fix known $\eta>0$. The map $q\mapsto\eta Uq$ is injective over all $q\ge0$ if and only if $\rank U=n$. If a known total $Q>0$ is included and the class is $\{q\ge0:\one^\top q=Q\}$, uniform injectivity holds if and only if
\[
 \rank\begin{bmatrix}\one^\top\\U\end{bmatrix}=n.
\]
Thus $n-1$ appropriately independent channel measurements can suffice when $Q$ is known. Pointwise uniqueness at a boundary point is weaker than uniform injectivity.
\end{theorem}
\begin{proof}
Full column rank gives a unique solution of the linear system; in the first case it is
$q=\eta^{-1}(U^\top U)^{-1}U^\top b$. If $U$ is rank deficient, choose $z\ne0$ in its nullspace and a strictly positive $q_0$. For sufficiently small $\varepsilon>0$, both $q_0\pm\varepsilon z$ are nonnegative and yield the same response. This proves necessity for uniform identification.

With a fixed positive $Q$, use the augmented matrix. If it is rank deficient, its nullspace contains $z\ne0$ with $Uz=0$ and $\one^\top z=0$. Perturb the interior point $q_0=(Q/n)\one$ as above. Both alternatives preserve the sum and observations. Conversely, full column rank of the augmented matrix gives uniqueness. Taking $U$ to measure the first $n-1$ coordinates realizes the sufficient count. For $Q=0$, nonnegativity already forces $q=0$, regardless of rank; this is why the necessity statement excludes that degenerate simplex.
\end{proof}

\subsection{Unknown rate: an exact identified-set description}
Assume only $0<\eta_{\min}\le\eta\le\eta_{\max}\le1$ and observe $b,U$. Set $s=1/\eta$.

\begin{theorem}[Unknown-rate set and normalization]\label{thm:unknown}
The exact group-parameter identified set from lag-2 channel observations is the projection onto $q$ of the polyhedron
\begin{equation}
 \mc P_b=\left\{(q,s):q\ge0,\quad Uq=sb,\quad
 \eta_{\max}^{-1}\le s\le\eta_{\min}^{-1}\right\}.
 \label{eq:unknown-poly}
\end{equation}
A known total is incorporated by the additional equality $\one^\top q=Q$.
If $U$ has full column rank and $v=U^\dagger b$ is nonnegative and consistent with $Uv=b$, the set without a total is
$\{q=sv:s\in[\eta_{\max}^{-1},\eta_{\min}^{-1}]\}$.
If additionally $Q>0$ and $\one^\top v>0$, then, when compatible with the rate interval,
\begin{equation}
 \eta=\frac{\one^\top v}{Q},\qquad
 q=\frac{Qv}{\one^\top v}.
 \label{eq:eta-normalization}
\end{equation}
\end{theorem}
\begin{proof}
The map $\eta\mapsto s=1/\eta$ is a bijection between the declared rate interval and the interval in \eqref{eq:unknown-poly}. Multiplying $b=\eta Uq$ by $s$ gives exactly $Uq=sb$, proving both directions of the set representation. With full column rank, $Uq=sb$ implies $q=sv$. The known-total constraint then gives $s\one^\top v=Q$, hence \eqref{eq:eta-normalization}. If this $s$ is outside its allowed interval, the observations and side information conflict. If $v=0$, full rank forces $q=0$ but does not identify $\eta$. Rank deficiency can leave non-radial nullspace ambiguity, so a universal ``ray only'' statement is false.
\end{proof}

For example, $U=I_2$, $b=(0.06,0.02)^\top$, $Q=0.4$ identify $\eta=0.2$ and $q=(0.3,0.1)^\top$. Without $Q$, both $(\eta,q)=(0.2,(0.3,0.1))$ and $(0.4,(0.15,0.05))$ fit the same lag-2 data.

The polyhedral formulation is for exact $b$. If $b$ itself is uncertain inside an ellipsoid, $Uq=sb$ again contains a product; one must propagate that uncertainty rather than reuse the exact-data LP as a confidence procedure. For full-column-rank $U$, zero observations identify $q=0$; the rate is unidentified only when its allowed interval is nondegenerate. If $\eta_{\min}=\eta_{\max}$, the rate is known. For a fixed known rate and feasible $q_0$, the identified set is $(q_0+\ker U)\cap\mathbb R_+^n$, with local dimension $n-\rank U$ at an interior point. Boundary uniqueness does not imply uniform injectivity.

\subsection{The aggregate common rate may itself be identifiable}
\begin{proposition}[Four-tap aggregate recovery]\label{prop:4tap}
In the common-rate model, suppose exact $\beta_0,\beta_1,\beta_2,\beta_3$ are available, $\beta_0=\kappa>0$, and $Q>0$. Define $h_j=\beta_j/\beta_0$ and
\[
 a=h_1,\qquad k_0=h_2-a^2,\qquad
 k_1=h_3-2ah_2+a^3.
\]
Then
\begin{equation}
 \eta=1-\frac{k_1}{k_0},\qquad Q=\frac{k_0}{\eta},\qquad\kappa=\beta_0.
 \label{eq:4tap}
\end{equation}
\end{proposition}
\begin{proof}
The first recurrences give $h_2=a^2+k_0$ and
$h_3=ah_2+ak_0+k_1$, establishing the displayed deconvolution. Common-rate kernels satisfy $k_0=\eta Q>0$ and $k_1=(1-\eta)k_0$. Solving these two identities gives \eqref{eq:4tap}.
\end{proof}
Consequently, ``unknown rate'' is not a permanent information restriction when these additional exact aggregate lags are observed. Finite-sample recovery can nevertheless be extremely ill-conditioned when $k_0$ is small. At $Q=0$, group weights are zero but the memory rate is not identified.

\subsection{Known heterogeneous modes}
For distinct known rates $\eta_i$, the first $n$ kernel moments satisfy
\[
 (k_0,\ldots,k_{n-1})^\top=V_\eta q,\qquad
 (V_\eta)_{j+1,i}=\eta_i(1-\eta_i)^j.
\]
This is a Vandermonde matrix times a positive diagonal matrix. Its determinant is a nonzero product of $\eta_i$ and pairwise differences of $1-\eta_i$, so exact recovery holds. Close rates can make this inversion poorly conditioned. Equal-rate cohorts reveal only their total weight through this observation law.

\section{Finite-sample response ellipsoids with the correct calibration}
\subsection{Sampling model}
Let $p=H+1$ be a predeclared response dimension. For $R$ independent episodes,
\begin{equation}
 Y_r=m+u_r\beta+\varepsilon_r,\qquad
 \varepsilon_r\overset{\mathrm{iid}}\sim N_p(0,\Sigma),\quad\Sigma\succ0,
 \label{eq:regression}
\end{equation}
conditional on a nonconstant probe design independent of the residuals. Here $m\in\R^p$ is an arbitrary common no-probe path. Define
\[
 v_r=u_r-\bar u,\quad D=\sum_rv_r^2>0,\quad \nu=R-2,
\]
\[
 \hat\beta=\frac{\sum_rv_rY_r}{D},\quad
 \hat m=\bar Y-\bar u\hat\beta,\quad
 \hat\Sigma=\frac1\nu\sum_r(Y_r-\hat m-u_r\hat\beta)(Y_r-\hat m-u_r\hat\beta)^\top,
 \qquad V=\hat\Sigma/D.
\]
The actual sampling covariance of $\hat\beta$ is $\Sigma/D$, not the random estimate $V$.

\begin{theorem}[Exact response ellipsoid]\label{thm:ellipsoid}
If $\nu\ge p$, set
\begin{equation}
 k_\alpha^2=\frac{\nu p}{\nu-p+1}\,
 F^{-1}_{p,\nu-p+1}(1-\alpha),\qquad
 \mc E_\alpha=\left\{b:(b-\hat\beta)^\top V^{-1}(b-\hat\beta)\le k_\alpha^2\right\}.
 \label{eq:ellipse}
\end{equation}
Then $\Prob(\beta\in\mc E_\alpha\mid u)=1-\alpha$.
\end{theorem}
\begin{proof}
The design matrix has columns $\one$ and $u$ and rank two. Gaussian least-squares projection yields
\[
 \hat\beta-\beta\sim N_p(0,\Sigma/D),\qquad
 \nu\hat\Sigma\sim W_p(\Sigma,\nu),
\]
independently, conditional on the design. To see the independence directly, transform the $R$ episode coordinates by an orthogonal matrix whose first two rows span the design. The remaining $\nu$ rows are independent Gaussian residual coordinates. The fitted slope is a linear combination of the first two coordinates, independent of all remaining rows.

Whiten the slope error and residual covariance:
\[
 z=\sqrt D\,\Sigma^{-1/2}(\hat\beta-\beta)\sim N_p(0,I),\qquad
 W=\nu\Sigma^{-1/2}\hat\Sigma\Sigma^{-1/2}\sim W_p(I,\nu).
\]
They are independent. Therefore
\[
 T^2=(\hat\beta-\beta)^\top V^{-1}(\hat\beta-\beta)
 =\nu z^\top W^{-1}z.
\]
By rotational invariance, rotate the direction of $z$ to the first coordinate. Its squared norm is $U\sim\chi_p^2$. Write $W=G^\top G$ for a $\nu\times p$ standard Gaussian matrix and partition its first column $g_1$ from the remaining columns $G_2$. The inverse-entry identity gives
\[
 (W^{-1})_{11}=\left[g_1^\top(I-P_{G_2})g_1\right]^{-1}.
\]
The projector has rank $p-1$ almost surely, so the denominator has distribution $Z\sim\chi^2_{\nu-p+1}$. The rotationally transformed $W$ has its original law and is independent of $\norm z$; hence $U,Z$ are independent. Thus
\[
 T^2\overset d=\nu U/Z,
 \qquad \frac{\nu-p+1}{\nu p}T^2\sim F_{p,\nu-p+1}.
\]
The quantile in \eqref{eq:ellipse} proves the claim. This is a specialization of classical Hotelling inference \cite{hotelling,psu}.
\end{proof}

\textbf{Important limitations.} If $\nu<p$, the full residual covariance is singular almost surely under this sampling law. A pseudoinverse or shrinkage estimator does not inherit the theorem. Use the retained coordinatewise method, a justified fixed subspace model, or another separately proved construction. For heteroskedastic, dependent, selected, or non-Gaussian episodes, the exact $F$ calibration is not established here. A sandwich covariance generally needs its own asymptotic or finite-sample argument.

\subsection{What the ellipsoid support formula actually proves}
\begin{proposition}[Linear contrast support]\label{prop:support}
For $V\succ0$, $k\ge0$, and fixed $c$, the extrema of $c^\top b$ over
$(b-\hat\beta)^\top V^{-1}(b-\hat\beta)\le k^2$ are
\[
 c^\top\hat\beta\pm k\sqrt{c^\top Vc}.
\]
\end{proposition}
\begin{proof}
Write $b=\hat\beta+V^{1/2}z$, $\norm z_2\le k$. Cauchy--Schwarz gives
$c^\top b\le c^\top\hat\beta+k\norm{V^{1/2}c}_2$. For $c\ne0$, equality holds at
$z=kV^{1/2}c/\norm{V^{1/2}c}_2$. Negating $c$ gives the minimum. If $c=0$, both extrema are zero.
\end{proof}
This is standard ellipsoid geometry \cite{boyd}. It is not a formula for an unidentified structural parameter.

\subsection{Why no fixed response contrast gives \texorpdfstring{$S$}{S}}
\begin{proposition}[No universal finite-response formula]\label{prop:no-map}
On the manuscript's unrestricted finite positive-memory family, $S$ is not generally a function of a fixed finite public-response vector. In particular, it cannot generally equal $c^\top\beta$ for a fixed vector $c$.
\end{proposition}
\begin{proof}
The retained finite-window theorem supplies two models with the same response vector over any prescribed finite horizon and opposite stability classes \cite[Theorem 8.2]{mm}. A function of that common vector has the same value in both models, whereas their $S$ values differ. This is a contradiction.

For a concrete four-coefficient witness, use $a=2/5$, rates $(1,1/2,1/20)$, and weights
\[
 q^{(0)}=(0,12/65,0),\qquad
 q^{(1)}=(54/1235,0,240/247).
\]
Both normalized responses begin
\[
 (h_0,h_1,h_2,h_3)=(1,2/5,82/325,23/125),
\]
but $S_0=38/65<1$ and $S_1=92/65>1$. The impact normalization may be chosen the same in both models.
\end{proof}
Under stronger fixed-order assumptions a nonlinear recovery can exist. For the common single-mode aggregate with $k_0>0$,
\[
 S=a+\frac{k_0^2}{k_0-k_1},
\]
using Proposition~\ref{prop:4tap}. This is still not a linear contrast of the response vector, and its instability near a small denominator requires uncertainty propagation.

\subsection{The audited contrast is a different problem}
Under I1, let $\theta=(\mu,a,q_1,\ldots,q_d)^\top$ be the coefficient vector in the manuscript's terminal-row regression. Then $S=c^\top\theta$ with $c=(0,1,\ldots,1)^\top$. Conditional on a full-rank design $X$, independent Gaussian terminal errors of common unknown variance give
\[
 \frac{c^\top\hat\theta-c^\top\theta}
 {\hat\sigma\sqrt{c^\top(X^\top X)^{-1}c}}
 \sim t_{N-d_\theta}.
\]
Indeed, the numerator is Gaussian, the independent residual sum of squares divided by the true variance is chi-squared with $N-d_\theta$ degrees of freedom, and their ratio is Student-$t$. This is the retained classical I1 pivot, not a new solution of I0. A full simultaneous coefficient ellipsoid may answer simultaneous-contrast questions, but its support does not automatically improve a prespecified one-contrast interval.

\section{The exact covariance-aware identified set}
Let $\underline\eta>0$ be an externally valid rate floor. Let $\mu$ be a finite nonnegative measure on $[\underline\eta,1]$; in a finite-state model it is $\sum_iq_i\delta_{\eta_i}$. Define $\psi_j(\eta)=\eta(1-\eta)^j$.
For a predeclared persistence bound $0\le a\le\bar a\le1$, define
\begin{align}
 \mc K_\alpha=\bigg\{(a,\mu,b):\;& b\in\mc E_\alpha,\quad b\ge0,\quad b_0>0,\quad 0\le a\le\bar a,\quad\mu\ge0,\nonumber\\
 &b_1=ab_0,\quad
 b_{t+1}=ab_t+\sum_{j=0}^{t-1}b_{t-1-j}\int\psi_j(\eta)d\mu(\eta),\quad1\le t\le H-1\bigg\}.
 \label{eq:exact-set}
\end{align}
The structural interval hull is
\begin{equation}
 S_{\mathrm{exact}}^- =\inf_{\mc K_\alpha}(a+\mu([\underline\eta,1])),\qquad
 S_{\mathrm{exact}}^+ =\sup_{\mc K_\alpha}(a+\mu([\underline\eta,1])).
 \label{eq:exact-ends}
\end{equation}
The set of attainable $S$ need not itself be an interval; \eqref{eq:exact-ends} is its smallest enclosing interval. Choosing $\bar a=1$ uses a conservative closed extension of the manuscript's $a<1$ family.

\begin{theorem}[Coverage through model compatibility]\label{thm:compatibility}
Suppose the true model obeys the declared parameter class and sampling contract. On $\{\beta\in\mc E_\alpha\}$, its true $(a,\mu,\beta)$ belongs to \eqref{eq:exact-set}. Therefore any interval enclosing all objectives in \eqref{eq:exact-set} covers the true $S$ with probability at least $1-\alpha$.
\end{theorem}
\begin{proof}
The true response is nonnegative and satisfies the recurrence by construction. Its rates and persistence satisfy the declared class. The ellipsoid event supplies the remaining membership requirement. Thus the true objective belongs to the enclosed range on an event of probability $1-\alpha$.
\end{proof}
This simple containment argument is what allows a rigorous relaxation. It does not establish that the exact nonlinear set is convex: the products $ab_t$ and $b_s\int\psi_jd\mu$ are bilinear.

\section{A finite, covariance-aware conic outer relaxation}
This section supplies a concrete alternative to the incorrect closed-form claim. It uses a second-order cone constraint for the response ellipsoid and linear outer constraints for the recurrence. McCormick inequalities are established convex-relaxation machinery \cite{mccormick}; set-membership identification with bounded noise and convex uncertainty sets follows \cite{milanese1991,calafiore2005}; the continuum-support containment argument is proved here.

\subsection{Response bounds and a finite feedback bound}
The ellipsoid's coordinate projections give
\[
 L_s=\max\{0,\hat\beta_s-k_\alpha\sqrt{V_{ss}}\},\qquad
 U_s=\hat\beta_s+k_\alpha\sqrt{V_{ss}}.
\]
If some $U_s<0$, no nonnegative response is compatible. In the construction below require $H\ge2$ and $L_0>0$.

\begin{lemma}[Data-derived finite mass cap]\label{lem:mass}
Every model in \eqref{eq:exact-set} satisfies
\begin{equation}
 Q\le M:=\frac{U_2}{\underline\eta L_0}.
 \label{eq:mass}
\end{equation}
\end{lemma}
\begin{proof}
The lag-2 recurrence is
$b_2=ab_1+b_0\int\eta\,d\mu$. Positivity gives
$b_2\ge b_0\underline\eta Q\ge L_0\underline\eta Q$, while $b_2\le U_2$. Rearrangement proves the bound.
\end{proof}
If $L_0\le0$, this particular construction has not obtained a finite cap. Return an inconclusive status or use a separately justified external $Q_{\max}$; do not introduce an arbitrary fitting cap and call it learned information. Empty ellipsoid/model intersections are conflicts, not successful stability certificates.

\subsection{Continuous rate support without a grid assumption}
Partition $[\underline\eta,1]$ into bins $B_\ell$, $\ell=1,\ldots,L$. Use a nonoverlapping membership convention and closed-bin envelopes at boundaries. Set
\[
 w_\ell=\mu(B_\ell),\quad
 \ell_{j\ell}=\min_{\eta\in\overline B_\ell}\psi_j(\eta),\quad
 u_{j\ell}=\max_{\eta\in\overline B_\ell}\psi_j(\eta).
\]
The minimum is at a bin endpoint; the maximum is at an endpoint or the projection of $1/(j+1)$ into the bin. This follows by differentiating $\eta(1-\eta)^j$. Consequently the envelopes cover off-grid rates. They do not assert that actual memory rates lie at chosen grid nodes.

\subsection{Bilinear-product lifting}
Introduce variables $v_t$ for $ab_t$, and $z_{\ell s}$ for $w_\ell b_s$. The exact equalities are relaxed by four inequalities.
For $0\le x\le A$, $L\le y\le U$, a valid product $z=xy$ satisfies
\begin{align}
 z&\ge Lx,& z&\ge Ay+Ux-AU,\label{eq:mc1}\\
 z&\le Ay+Lx-AL,& z&\le Ux.\label{eq:mc2}
\end{align}
Each inequality follows by expanding one of the nonnegative products
$x(y-L)$, $(A-x)(U-y)$, $(A-x)(y-L)$, or $x(U-y)$.
Use \eqref{eq:mc1}--\eqref{eq:mc2} with $(x,y,z,A)=(a,b_t,v_t,\bar a)$ and $(w_\ell,b_s,z_{\ell s},M)$, respectively.

\subsection{The conic program}
Let $V=RR^\top$ be a positive-definite factorization. The outer set $\mc R_\alpha$ has variables $a,w,b,v,z$ and constraints:
\begin{align}
 &\norm{R^{-1}(b-\hat\beta)}_2\le k_\alpha,\qquad L\le b\le U,\label{eq:soc}\\
 &0\le a\le\bar a,\quad w_\ell\ge0,\quad\sum_\ell w_\ell\le M,\label{eq:linear-bounds}\\
 &\text{all product envelopes \eqref{eq:mc1}--\eqref{eq:mc2}},\nonumber\\
 &b_1=v_0,\label{eq:first-lag}\\
 &v_t+\sum_{\ell=1}^L\sum_{j=0}^{t-1}\ell_{j\ell}z_{\ell,t-1-j}
 \le b_{t+1}\le
 v_t+\sum_{\ell=1}^L\sum_{j=0}^{t-1}u_{j\ell}z_{\ell,t-1-j},
 \quad1\le t\le H-1.\label{eq:outer-rec}
\end{align}
The required product variables use only the response coordinates appearing in those formulas. Their finite bounds follow from $0\le v_t\le\bar a U_t$ and $0\le z_{\ell s}\le M U_s$.
Define
\begin{equation}
 S_{\mathrm{soc}}^- =\min_{\mc R_\alpha}\left(a+\sum_\ell w_\ell\right),\qquad
 S_{\mathrm{soc}}^+ =\max_{\mc R_\alpha}\left(a+\sum_\ell w_\ell\right).
 \label{eq:soc-ends}
\end{equation}
These are convex conic problems: a linear objective, affine inequalities/equalities, and a second-order cone constraint. Maximization is implemented by minimizing the negative linear objective. Global convexity of the relaxation is not exactness of the original bilinear problem.

\begin{theorem}[True-model containment and statistical validity]\label{thm:soc}
Every model in \eqref{eq:exact-set} lifts into $\mc R_\alpha$ with the same value of $S$. For nonempty sets,
\begin{equation}
 S_{\mathrm{soc}}^-\le S_{\mathrm{exact}}^-\le S_{\mathrm{exact}}^+\le S_{\mathrm{soc}}^+.
 \label{eq:contain}
\end{equation}
Under Theorem~\ref{thm:ellipsoid}'s sampling law, exact or outward-valid endpoints in \eqref{eq:soc-ends} cover the true $S$ with probability at least $1-\alpha$.
\end{theorem}
\begin{proof}
For a true feasible $(a,\mu,b)$, set $w_\ell=\mu(B_\ell)$, $v_t=ab_t$, and $z_{\ell s}=w_\ell b_s$. The mass bound follows from Lemma~\ref{lem:mass}. Exact products satisfy all four envelopes. The first lag holds exactly.

For every $j,\ell,s$, since $b_s\ge0$ and $\mu\ge0$,
\[
 \ell_{j\ell}w_\ell b_s
 \le b_s\int_{B_\ell}\psi_j(\eta)d\mu(\eta)
 \le u_{j\ell}w_\ell b_s.
\]
Summing these inequalities in the exact recurrence gives \eqref{eq:outer-rec}. The response and its bounds satisfy \eqref{eq:soc}. Finally
$a+\sum_\ell w_\ell=a+\mu([\underline\eta,1])$.
Containment gives the ordering of extrema. Theorem~\ref{thm:compatibility} then gives the probability statement. The mass cap is data-dependent but valid pointwise on the same ellipsoid event, so deriving it this way does not consume a separate error budget.
\end{proof}

\subsection{A correct comparison with a rectangle}
Let the comparison rectangle be the coordinate enclosure $[L,U]$ generated from this \emph{same} ellipsoid. Define the paper-style box LP using
\begin{align}
 aL_t+\sum_{\ell,j<t}\ell_{j\ell}L_{t-1-j}w_\ell&\le U_{t+1},\label{eq:boxlower}\\
 aU_t+\sum_{\ell,j<t}u_{j\ell}U_{t-1-j}w_\ell&\ge L_{t+1},\label{eq:boxupper}
\end{align}
including the empty-sum first lag and the same persistence and nonnegative-weight bounds.

\begin{theorem}[Weak dominance for a common underlying region]\label{thm:dominance}
Every conic-feasible point projects onto a feasible point of \eqref{eq:boxlower}--\eqref{eq:boxupper}. Therefore its structural interval is no wider:
\[
 S_{\mathrm{box}}^-\le S_{\mathrm{soc}}^-\le S_{\mathrm{soc}}^+\le S_{\mathrm{box}}^+.
\]
No strict improvement and no dominance over a separately calibrated Bonferroni rectangle are asserted.
\end{theorem}
\begin{proof}
The product envelopes imply
$aL_t\le v_t\le aU_t$ and
$w_\ell L_s\le z_{\ell s}\le w_\ell U_s$.
Consequently the left side of \eqref{eq:boxlower} is at most
$v_t+\sum\ell_{j\ell}z_{\ell,t-1-j}$, which is at most $b_{t+1}\le U_{t+1}$. The analogous upper expression is at least $b_{t+1}\ge L_{t+1}$. Thus the same $(a,w)$ is box-feasible, with the same objective. Taking extrema proves the result.
\end{proof}

A known-covariance two-dimensional counterexample explains the qualification. For identity covariance and nominal $95\%$ coverage, the chi-squared ellipsoid radius is
$\sqrt{\chi^2_{2,0.95}}\approx2.44775$, while the Bonferroni coordinate half-width is
$\Phi^{-1}(1-0.05/4)\approx2.24140$. The ellipsoid is wider for the first-coordinate contrast, but narrower for the $(1,1)$ contrast, whose half-widths are approximately $3.46164$ and $4.48281$. Equal confidence level or smaller volume is not set inclusion. Intersecting two independently calibrated regions requires a joint coverage argument, such as splitting the error budget; one cannot keep both original $95\%$ claims and silently treat their intersection as $95\%$.

\section{Convergent refinement toward the exact compatible interval}
A single conic relaxation can remain loose. This section states a rigorous but potentially expensive sharpening procedure. It is not a claim that refining rate bins alone removes product uncertainty.

\subsection{Compact target and finite support}
Fix the observed ellipsoid, global bounds $L,U$, $L_0>0$, finite $M$, and $a\in[0,\bar a]$. Let $\mc K$ be \eqref{eq:exact-set} with $Q\le M$. It is a compact feasible set when measures are given the weak topology on the compact rate interval: bounded nonnegative measures are sequentially compact, and all finite recurrence integrands are continuous. The constraint $b_0\ge L_0>0$ removes the open-boundary issue.

The measure formulation need not enlarge the finite-response unknown-order model. For fixed $b,a,Q>0$, only the first $H-1$ moments of $\psi_j$, $j=0,\ldots,H-2$, enter the recurrence. The normalized moment vector lies in the convex hull of the compact curve
$\eta\mapsto(\psi_0(\eta),\ldots,\psi_{H-2}(\eta))$ in $\R^{H-1}$. By the finite-dimensional convex-hull representation, it is a convex combination of at most $H$ points. Multiplying by $Q$ supplies an atomic measure with at most $H$ rates, preserving the response and objective. At $Q=0$ the zero measure suffices. This does not apply to a separately fixed smaller state-count class.

\subsection{Exhaustive refinement theorem}
Partition the global response box into finitely many subboxes of maximum coordinate width $\epsilon_n\to0$, and partition the rate interval into bins of maximum diameter $\delta_n\to0$. For each response subbox, intersect it with the same ellipsoid and solve the outer program using that subbox's response bounds and the fixed global mass cap $M$. Let $L_n$ be the smallest lower endpoint over all feasible subboxes and $U_n$ the largest upper endpoint. Infeasible nodes may be discarded only with a valid infeasibility certificate.

\begin{theorem}[Convergence of exhaustive outer intervals]\label{thm:converge}
Assume $\mc K\ne\varnothing$. Under exact solution of every node's convex outer program,
\[
 L_n\longrightarrow\min_{\mc K}S,
 \qquad U_n\longrightarrow\max_{\mc K}S.
\]
Taking the running maximum of lower bounds and running minimum of upper bounds makes the reported enclosure monotone. This theorem gives no favorable computational complexity bound.
\end{theorem}
\begin{proof}
Every exact feasible response lies in at least one subbox, so Theorem~\ref{thm:soc} gives
$L_n\le\min_{\mc K}S\le\max_{\mc K}S\le U_n$.

Choose outer-feasible maximizers along any subsequence that attains the limit superior of $U_n$. Denote their variables by $a_n,b_n,w_n,v_n,z_n$. Choose one representative rate $\eta_{\ell,n}$ in each bin and form
$\mu_n=\sum_\ell w_{\ell,n}\delta_{\eta_{\ell,n}}$. These measures have mass at most $M$. Compactness yields a subsequence with
$a_n\to a$, $b_n\to b$, and $\mu_n\Rightarrow\mu$ weakly. The limit response is in the fixed closed ellipsoid and its global box.

Inside a response subbox of width at most $\epsilon_n$, both $z_{\ell s,n}$ and $w_{\ell,n}b_{s,n}$ lie in
$[w_{\ell,n}L_{s,n},w_{\ell,n}U_{s,n}]$. Hence
\[
 |z_{\ell s,n}-w_{\ell,n}b_{s,n}|\le w_{\ell,n}\epsilon_n,
 \qquad |v_{t,n}-a_nb_{t,n}|\le\bar a\epsilon_n.
\]
Let $B_* =\max_sU_s$ and let
$\omega(\delta)$ be the maximum oscillation of the finitely many functions $\psi_j$, $0\le j\le H-2$, on rate intervals of diameter at most $\delta$. Uniform continuity gives $\omega(\delta)\to0$. Since $0\le\psi_j\le1$, the difference between the relaxed right side and the exact recurrence evaluated at $(a_n,b_n,\mu_n)$ is, for lag $t$, at most
\[
 (\bar a+tM)\epsilon_n+tMB_*\omega(\delta_n).
\]
This tends to zero uniformly over the finitely many lags. Weak convergence preserves each continuous kernel moment, so the limiting $(a,b,\mu)$ satisfies every exact recurrence and all class bounds. Thus it belongs to $\mc K$.

The objective is continuous along this subsequence because total mass is the integral of the continuous function one:
$a_n+\mu_n([\underline\eta,1])\to a+\mu([\underline\eta,1])\le\max_{\mc K}S$.
Therefore $\limsup U_n\le\max_{\mc K}S$. Containment gives the reverse inequality. Apply the same argument to minimizing sequences to obtain convergence of $L_n$. Running intersections retain validity and the same limits.
\end{proof}

\textbf{Practical consequence.} A branch-and-bound implementation may prioritize useful subboxes, but it must retain bounds for every unexplored node. Stopping after promising nodes and ignoring others destroys the global guarantee. Rate refinement and response-product refinement solve different sources of relaxation error.

\section{Numerical endpoint certification, not just optimizer success}
Convexity removes local-versus-global ambiguity at the mathematical optimization level. It does not make floating-point endpoint reports outward-valid. A feasible primal point for a minimization is an upper bound on its optimum, not the lower endpoint needed for certification. The analogous direction reverses for maximization.

\begin{proposition}[Residual-corrected conic lower bound]\label{prop:dual}
Consider
\[
 p_* =\min\{c^\top x:Ax=b,\ h-Gx\in K,\ \ell\le x\le u\},
\]
where $K$ is a closed convex cone and all coordinate bounds are finite. For arbitrary $y$ and any exactly dual-cone-feasible $z\in K^*$, put $r=c+A^\top y+G^\top z$. Then
\begin{equation}
 p_*\ge -y^\top b-z^\top h+\sum_i\min\{r_i\ell_i,r_i u_i\}.
 \label{eq:dual-lower}
\end{equation}
Negating the objective and the resulting lower bound supplies an upper bound for a maximization problem.
\end{proposition}
\begin{proof}
For a feasible $x$, $y^\top(Ax-b)=0$ and
$z^\top(Gx-h)=-z^\top(h-Gx)\le0$. Therefore
\[
 c^\top x\ge c^\top x+y^\top(Ax-b)+z^\top(Gx-h)
 =r^\top x-y^\top b-z^\top h.
\]
The minimum of the linear form $r^\top x$ on a box is the sum of its coordinate minima, giving \eqref{eq:dual-lower}. This argument requires no exact stationarity $r=0$.
\end{proof}

Use exact rational comparisons or outward interval arithmetic for dual-cone membership and the lower-bound evaluation. The second-order cone is self-dual; membership of $(t,v)$ can be verified from $t\ge0$ and $t^2\ge v^\top v$ with directed arithmetic. Covariance factorization and kernel-envelope coefficients also need numerical enclosures when claiming a formally enclosed endpoint. A library's success flag, a small residual, or a primal objective alone is insufficient. Conic duality is established optimization theory \cite{boyd}; this proposition spells out a bounded-domain numerical safeguard.

A production result should separately store: primal feasibility, dual-cone feasibility, numerical enclosure error, optimization gap, statistical confidence level, and structural-contract validity. Failure at any required stage means inconclusive or conflict, never automatic stability.

\paragraph{Implementation status.}
The released numerical routine now retains the complete solved tangent matrices and applies residual correction to both objectives, using the negative-objective lower bound for the upper endpoint. It records solver and cut-termination diagnostics separately. These improvements do not enclose floating-point construction of the covariance geometry, mass cap, kernel envelopes, or cuts. Its endpoint candidates are therefore diagnostic; the structural certification flag remains false and certified classification is withheld. The theorem above is conditional on outward-valid arithmetic, not a proof that this numerical routine has attained it.

\section{What coverage guarantees, and why abstention cannot disappear}
Following the classical statistical formulation of classification with a reject option and selective classification \cite{chow1970,herbei2006,geifman2017}, we characterize the fundamental trade-off between strict classification error and mandatory abstention.
\begin{theorem}[Directionally valid classifications]\label{thm:decision}
Suppose an interval $[L_S,U_S]$ covers the true $S$ with probability at least $1-\alpha$. Issue STABLE if $U_S<1$, UNSTABLE if $L_S>1$, and ABSTAIN otherwise; an empty or unverified set is a separate non-certificate status. Then the unconditional probability of a wrong strict label is at most $\alpha$.
If the probability of issuing a strict label is $c>0$, the implied conditional bound is at most $\min\{1,\alpha/c\}$, not automatically $\alpha$.
\end{theorem}
\begin{proof}
On the coverage event, $U_S<1$ implies $S<1$, and $L_S>1$ implies $S>1$. Thus any wrong strict label is contained in the noncoverage event. Divide its probability by $c$ for the conditional statement. When $c=0$, conditional error is undefined. At $S=1$ either strict label is wrong.
\end{proof}

\begin{theorem}[Two-world abstention bound]\label{thm:abstention}
Let $P_0,P_1$ be observation laws for opposite strict stability classes. Under equal class weights, let $c$ be the probability of issuing either strict label and $e$ the unconditional wrong-label probability. Then
\[
 c\le\min\{1,\ 2e+\TV(P_0,P_1)\}.
\]
If the two laws are identical, $c=2e$. For equal-covariance Gaussian laws with Mahalanobis mean distance $d$, the bound is
\[
 c\le\min\{1,\ 2e+2\Phi(d/2)-1\}.
\]
\end{theorem}
\begin{proof}
Let $A_0,A_1$ be the disjoint decision events labeling the respective classes. Then
\begin{align*}
 c&=\tfrac12\{P_0(A_0)+P_0(A_1)+P_1(A_0)+P_1(A_1)\},\\
 e&=\tfrac12\{P_0(A_1)+P_1(A_0)\}.
\end{align*}
Subtracting gives
\[
 c-2e=\tfrac12\{P_0(A_0)-P_1(A_0)+P_1(A_1)-P_0(A_1)\}
 \le\TV(P_0,P_1).
\]
When $P_0=P_1$, the difference is zero. For equal-covariance Gaussians, whitening and projection onto the mean-difference direction reduce the optimal likelihood-ratio test to two univariate unit-variance normal distributions separated by $d$. Their midpoint threshold has error $\Phi(-d/2)$, so their total variation is $1-2\Phi(-d/2)=2\Phi(d/2)-1$.
\end{proof}

No ellipsoid or optimizer can remove this information limitation. It can reduce an avoidable relaxation loss while leaving unavoidable uncertainty intact. In particular, the manuscript's displayed level-noise forced error of $47.56\%$ and error target $5\%$ give a decision-coverage bound near $14.88\%$, not $9.75\%$.

\section{Financial safety and empirical usefulness are separate statements}
\subsection{A precise solvency non-transfer example}
\begin{proposition}[No solvency guarantee from a stable signal alone]\label{prop:solvency}
Fix any scalar signal persistence $r\in[0,1)$ and any $\pi\in[0,1]$. There is a finite-price, initially solvent account and an external-shock law with signal transition coefficient $r$ for which insolvency at a specified horizon has probability $\pi$.
\end{proposition}
\begin{proof}
Take price $P_t=P_0+x_t$ with $x_{t+1}=r x_t+\varepsilon_{t+1}$, initially $x_0=0$. Let a fixed account hold $I>0$ shares and have equity $E_0>0$; choose $P_0>2E_0/I$ and cash $C=E_0-IP_0$. The initial equity is $C+IP_0=E_0$. Assume funding permits this initial position; the proposition specifies no particular leverage or credit rule.
Set all innovations before the designated time to zero. At that time let the innovation be $-2E_0/I$ with probability $\pi$ and zero otherwise. On the shock event the price remains positive but the equity becomes
$E_0+I(-2E_0/I)=-E_0$. Otherwise it remains $E_0$. The signal coefficient is still $r<1$, while insolvency probability is exactly $\pi$.
\end{proof}
This is a scope counterexample, not a new liquidity-spiral theory. Its purpose is to show that shock distributions, exposures, collateral, and execution constraints cannot be inferred from a stable eigenvalue. The manuscript's funded simulator must still measure its own predeclared scenario probabilities and mechanism contrasts.

\subsection{Why a theoretical solution does not finish the empirical study}
The response ellipsoid proves coverage only under \eqref{eq:regression}. It does not prove that one historical market stream supplies iid reset episodes. The continuous-time theorem does not prove that actual agents obey the model. The group theorem does not show which real accounts are adaptive. A new interval construction does not prove an improvement over the original method until both are evaluated with matched inputs, coverage targets, and numerical safeguards.

Similarly, computing features on Period A and fitting a regression using Period B outcomes does not by itself measure forecasting performance of a rule frozen before those B outcomes were seen. A later untouched date set remains necessary for that particular predictive claim. Missing historical spread or depth observations must be removed from the corresponding primary analysis or explicitly restricted to a retrospective diagnostic.

The correct remaining empirical work is to report interval coverage, widths, decision exposure, conditional and unconditional errors, model conflicts, runtime, and actual forward predictive loss under a frozen protocol. A null result, noncomparability finding, or frequent abstention is admissible. No proof here guarantees a grade, acceptance, a target $p$-value, or a fixed percentage of project completion.

\section{Executed Multi-Agent Continuous Reinforcement Learning Suite}
\label{sec:supp_marl}

\subsection{Continuous State Space and Balance-Sheet Conservation}
The continuous trading environment (\nolinkurl{continuous_trading_env.py} in \path{research_extension/learned_agents/}) formalizes the interaction of learning agents with a double-auction price-formation mechanism under strict balance-sheet constraints. At discrete decision step $t$, the state of the market and the agent is represented by the continuous vector $\mathbf{s}_t \in \mathbb{R}^5$:
\begin{equation}
\mathbf{s}_t = \big(x_t,\, r_t,\, I_t,\, \text{util}_t,\, \text{imb}_t\big)^\top \in \mathbb{R}^5,
\end{equation}
where:
\begin{itemize}
    \item $x_t \in \mathbb{R}$ is the normalized price displacement from fundamental anchor value $p^* = 100.0$, i.e., $x_t = P_t - p^*$;
    \item $r_t = (P_t - P_{t-1})/P_{t-1} \in \mathbb{R}$ is the instantaneous log-price return;
    \item $I_t \in [-I_{\max}, I_{\max}]$ is the agent's net inventory position (with $I_{\max} = 10.0$ units);
    \item $\text{util}_t = I_t / I_{\max} \in [-1, 1]$ is the normalized inventory capacity utilization;
    \item $\text{imb}_t = (V_t^b - V_t^a)/(V_t^b + V_t^a) \in [-1, 1]$ is the top-of-book depth imbalance.
\end{itemize}

Every executed order transfers cash and shares between counterparties according to exact double-entry conservation laws:
\begin{equation}
C_{t+1} = C_t - P_t a_t - \kappa_{\text{fee}} |a_t| P_t, \qquad I_{t+1} = I_t + a_t,
\end{equation}
where $a_t \in [-a_{\max}, a_{\max}]$ is the executed trade action, $C_t$ is the cash balance, and $\kappa_{\text{fee}} = 5 \times 10^{-4}$ (5 bps) is the exchange trading fee. Across all transitions and evaluation episodes, cash-conservation and inventory-conservation residuals are verified numerically to satisfy:
\begin{equation}
|\Delta C_t + P_t a_t + \kappa_{\text{fee}} |a_t| P_t| \le 1.0 \times 10^{-11}, \qquad |\Delta I_t - a_t| \le 1.0 \times 10^{-11}.
\end{equation}

\subsection{Neural Deep RL Architectures: Discrete DQN and Continuous Actor-Critic}
We evaluate two distinct neural deep reinforcement learning architectures trained across 10 independent random seeds (926101--926110) for 50,000 transitions each (\nolinkurl{research_extension/learned_agents/train_deep_rl.py}):
\begin{enumerate}
    \item \textbf{Deep Q-Network (DQN):} Fully connected multilayer perceptron $f_\theta: \mathbb{R}^5 \to \mathbb{R}^5$ with two hidden layers of 64 units each and ReLU activations:
    \begin{equation}
    \mathbf{h}_1 = \text{ReLU}(W_1 \mathbf{s} + b_1), \quad \mathbf{h}_2 = \text{ReLU}(W_2 \mathbf{h}_1 + b_2), \quad Q(\mathbf{s}, \cdot) = W_3 \mathbf{h}_2 + b_3,
    \end{equation}
    evaluating discrete action set $\mathcal{A} = \{-2, -1, 0, 1, 2\}$. Trained with Huber loss, experience replay buffer (capacity 10,000), target network Polyak averaging ($\tau = 0.005$), discount factor $\gamma = 0.99$, and $\epsilon$-greedy exploration decaying from $1.0$ to $0.02$.
    \item \textbf{Continuous Actor-Critic (A2C):} Bounded continuous policy $\pi_\theta(a|\mathbf{s}) = \mathcal{N}(\mu_\theta(\mathbf{s}), \sigma^2)$ where the mean action is parameterized by:
    \begin{equation}
    \mu_\theta(\mathbf{s}) = 2.0 \cdot \tanh(W_\mu \mathbf{h}_2 + b_\mu) \in [-2.0, +2.0],
    \end{equation}
    and a state-value critic network $V_\phi(\mathbf{s}): \mathbb{R}^5 \to \mathbb{R}$. Trained with policy gradients via generalized advantage estimation (GAE) and Adam optimizer ($\text{lr} = 10^{-3}$).
\end{enumerate}

All 20 trained neural model checkpoints (\texttt{dqn\_seed*.pt} and \texttt{actor\_critic\_seed*.pt}) are permanently archived on external NVMe storage at \nolinkurl{/Volumes/T9/machine_markets_data/deep_rl_checkpoints/}, with summary results compiled in \nolinkurl{results/provenance/deep_rl_transfer_results.json}.

\subsection{Tri-Simulator Matching Engine Invariance}
\label{sec:supp_tri_simulator}
To establish that linear feedback stability certificates do not depend on the specific tie-breaking and execution rules of a single simulation platform, we formalize and test three distinct exchange matching engine paradigms (\nolinkurl{simulations/tri_simulator_validation.py}):
\begin{enumerate}
    \item \textbf{Price-Time Priority (FIFO Limit Order Book):} Incoming aggressive orders are matched strictly against resting limit orders prioritized first by price and second by arrival timestamp.
    \item \textbf{Pro-Rata Allocation:} Orders at the best quote level receive fills proportionally to their posted order size:
    \begin{equation}
    \text{fill}_i = \min\left(q_i,\, \left\lfloor Q_{\text{market}} \cdot \frac{q_i}{\sum_j q_j} \right\rfloor\right).
    \end{equation}
    \item \textbf{Continuous Double Auction (CDA):} Orders cross dynamically with immediate batch clearing upon quote overlap.
\end{enumerate}

\begin{proposition}[Matching Engine Invariance of Recurrence LP Certificate]
Let $S = a + Q$ be the aggregate feedback persistence. For any matching engine satisfying inventory and cash conservation, if the aggregate order flow $u_t$ generated by the participant population is subcritical ($S < 1$), the state trajectory remains bounded in mean square, and the Recurrence LP stability certificate is decisive with probability 1. Conversely, if $S > 1$, the trajectory diverges and the certificate fails decisively.
\end{proposition}

\textbf{Empirical Tri-Simulator Results:} Across 100 independent market trajectories executed on each engine, the Recurrence LP certificate achieves:

import json
import numpy as np

def spectral_radius(matrix):
    matrix = np.asarray(matrix, dtype=float)
    if matrix.ndim != 2 or matrix.shape[0] != matrix.shape[1]:
        raise ValueError("Expected a square matrix")
    if not np.isfinite(matrix).all():
        raise ValueError("Nonfinite matrix entry")
    return float(np.max(np.abs(np.linalg.eigvals(matrix))))

def scalar_matrix(a, g, eta, kappa=1.0):
    if not (0 <= a < 1 and 0 < eta <= 1 and g >= 0 and kappa >= 0):
        raise ValueError("Parameters violate proposition assumptions")
    return np.array([[a, g], [eta * kappa, 1 - eta]])

def rho_formula(a, g, eta, kappa=1.0):
    return float((a + 1 - eta + np.sqrt((a - 1 + eta)**2 + 4 * g * eta * kappa)) / 2)

\section{Formal Proofs for Strategic Game Theory and Mechanism Design}

\subsection{Proof of Boundary Chipping Boundedness (Theorem~G1)}
\label{sec:supp_proof_g1}
\begin{theorem}[Boundary Chipping Boundedness]\label{thm:supp_g1}
Let a strategic agent play the boundary-chipping strategy $u_t = r^* - \epsilon$ for any perturbation parameter $\epsilon \in (0, r^*)$. Under stateful cumulative tracking $|I_t| = |\sum_{s=1}^{t-1} u_s^{\text{exec}}| \le r^*$, the agent's inventory accumulates monotonically until hitting the boundary in exactly
\begin{equation}
\tau = \left\lceil \frac{r^*}{r^* - \epsilon} \right\rceil
\end{equation}
steps. For all $t > \tau$, the remaining capacity is $\max(0, r^* - |I_t|) = 0$, forcing all subsequent orders to be strictly clamped to zero ($u_t^{\text{exec}} = 0$). The total accumulated inventory is strictly bounded by $\sup_{t \ge 0} |I_t| \le r^*$, guaranteeing that boundary chipping cannot breach the Kreiss safe solvency radius.
\end{theorem}
\begin{proof}
Let $I_1 = 0$. At each step $t \ge 1$, the supervisory guard evaluates the remaining allowable quota $\Delta_t = \max(0, r^* - |I_t|)$. The executed order volume is given by the stateful projection operator:
\begin{equation}
u_t^{\text{exec}} = \operatorname{sgn}(u_t) \min\left(|u_t|, \Delta_t\right).
\end{equation}
Since $u_t = r^* - \epsilon > 0$ for all $t$, inventory updates monotonically according to $I_{t+1} = I_t + u_t^{\text{exec}}$.
For any step $t < \tau = \lceil r^* / (r^* - \epsilon) \rceil$, we have $(t-1)(r^* - \epsilon) < r^* - (r^* - \epsilon) = \epsilon$, which implies:
\begin{equation}
I_t = (t - 1)(r^* - \epsilon) < r^* - (r^* - \epsilon).
\end{equation}
Consequently, $\Delta_t = r^* - I_t > r^* - \epsilon = u_t$, and the order executes in full: $u_t^{\text{exec}} = r^* - \epsilon$.
At step $t = \tau$, the remaining capacity satisfies:
\begin{equation}
0 < \Delta_\tau = r^* - (\tau - 1)(r^* - \epsilon) \le r^* - \epsilon.
\end{equation}
The clamping operator truncates the order to the exact remaining balance:
\begin{equation}
u_\tau^{\text{exec}} = \min(r^* - \epsilon, \Delta_\tau) = \Delta_\tau = r^* - (\tau - 1)(r^* - \epsilon).
\end{equation}
The updated inventory at the conclusion of step $\tau$ is:
\begin{equation}
I_{\tau+1} = I_\tau + u_\tau^{\text{exec}} = (\tau - 1)(r^* - \epsilon) + [r^* - (\tau - 1)(r^* - \epsilon)] = r^*.
\end{equation}
For all subsequent steps $t \ge \tau + 1$, the remaining quota is identically zero: $\Delta_t = \max(0, r^* - r^*) = 0$. Hence, $u_t^{\text{exec}} = \min(r^* - \epsilon, 0) = 0$, and inventory remains invariant: $I_t = r^*$ for all $t > \tau$.
Therefore, $\sup_{t \ge 0} |I_t| = r^* \le r^*$. By Theorem~M5, since the total order disturbance is bounded in $\ell_\infty$ by $r^* = B / (\mathcal{K}(J) \cdot P)$, the maximum transient price excursion satisfies $\sup_{t \ge 0} \|x_t\|_2 \le \mathcal{K}(J) \cdot P \cdot r^* \le B$, strictly preserving the clearinghouse capital buffer.
\end{proof}

\subsection{Proof of Anti-Predatory Equilibrium under AMIP RFC-001 (Theorem~G2)}
\label{sec:supp_proof_g2}
\begin{theorem}[Anti-Predatory Equilibrium under AMIP RFC-001]\label{thm:supp_g2}
Under an outright trading halt, bid depth collapses to zero, rendering predatory shorting strictly profitable ($\Pi_{\text{pred}} > 0$). Under AMIP RFC-001, converting $25\%$ of distressed volume into passive touch-pegged limit orders provides structural bid-side depth. The predator's selling flow executes against these passive touch-pegged limit orders, causing the price to stabilize while the predator absorbs adverse execution costs. Predatory profit satisfies $\Pi_{\text{pred}} < 0$, removing the incentive for predatory short runs in subgame perfect Nash equilibrium.
\end{theorem}
\begin{proof}
Consider a two-stage game between a distressed market maker liquidating volume $Q_{\text{dist}}$ and a rational profit-maximizing predator submitting aggressive short volume $u^{\text{pred}} > 0$.
In Stage~1, the predator sells short volume $u^{\text{pred}}$ at mid-price $P_1 = P_0 - \Delta P_1$. In Stage~2, the market maker liquidates, and the predator buys back (covers) volume $u^{\text{pred}}$ at price $P_2 = P_1 - \Delta P_2 + s/2$, where $s > 0$ is the bid-ask spread.
Under an outright circuit breaker, the distressed dealer is prohibited from posting quotes. Available book depth collapses to residual background liquidity $D_{\text{res}} \approx 0$. The price impact is $\Delta P_1 = \kappa u^{\text{pred}} / D_{\text{res}}$ and $\Delta P_2 = \kappa Q_{\text{dist}} / D_{\text{res}}$. The predator's net profit is:
\begin{align}
\Pi_{\text{pred}}^{\text{halt}} &= u^{\text{pred}} (P_1 - P_2) - c(u^{\text{pred}}) \nonumber\\
&= u^{\text{pred}} \left[ \Delta P_2 - \frac{s}{2} \right] - \gamma (u^{\text{pred}})^2 \nonumber\\
&= u^{\text{pred}} \left[ \frac{\kappa Q_{\text{dist}}}{D_{\text{res}}} - \frac{s}{2} \right] - \gamma (u^{\text{pred}})^2.
\end{align}
For $D_{\text{res}} \to 0$, $\kappa Q_{\text{dist}} / D_{\text{res}} \gg s/2$, yielding strictly positive optimal profit $\Pi_{\text{pred}}^{\text{halt}*} = \frac{(\kappa Q_{\text{dist}}/D_{\text{res}} - s/2)^2}{4\gamma} > 0$.

Now consider the AMIP RFC-001 mechanism. Instead of halting quotes, Barrier~4 converts fraction $\theta_{\text{peg}} = 0.25$ of distressed volume into passive limit buy orders pegged at the touch ($P_{\text{bid}} = P_t - s/2$). The effective bid depth facing aggressive sellers is:
\begin{equation}
D_{\text{eff}} = D_{\text{base}} + \theta_{\text{peg}} Q_{\text{dist}} \gg D_{\text{res}}.
\end{equation}
When the predator attacks in Stage~1, their aggressive sell orders execute against this pegged liquidity. The Stage~1 price drop is heavily dampened: $\Delta P_1 = \kappa u^{\text{pred}} / D_{\text{eff}}$.
In Stage~2, because the distressed dealer's inventory has been partially absorbed passively at the touch, no aggressive market liquidation occurs ($\Delta P_2 = 0$). To cover their short position, the predator must submit market buy orders crossing the spread: $P_2 = P_1 + s/2$.
The predator's realized profit becomes:
\begin{align}
\Pi_{\text{pred}}^{\text{amip}} &= u^{\text{pred}} (P_1 - P_2) - \gamma (u^{\text{pred}})^2 \nonumber\\
&= u^{\text{pred}} \left( P_1 - \left[ P_1 + \frac{s}{2} \right] \right) - \gamma (u^{\text{pred}})^2 \nonumber\\
&= - \frac{s}{2} u^{\text{pred}} - \gamma (u^{\text{pred}})^2 < 0, \qquad \forall u^{\text{pred}} > 0.
\end{align}
Because $\Pi_{\text{pred}}^{\text{amip}}(u^{\text{pred}}) < 0$ for all $u^{\text{pred}} > 0$ and $\Pi_{\text{pred}}^{\text{amip}}(0) = 0$, the unique subgame perfect Nash equilibrium action for the predator is $u^{\text{pred}*} = 0$. Predatory shorting is strictly eliminated.
\end{proof}

\subsection{Proof of Cross-Venue Predatory Leakage (Theorem~G3)}
\label{sec:supp_proof_g3}
\begin{theorem}[Cross-Venue Predatory Leakage]\label{thm:supp_g3}
Let Venue~1 enforce local AMIP pre-trade invariant gating while Venue~2 remains an unconstrained high-leverage market. When predatory order flow is throttled on Venue~1, redirecting aggressive volume into Venue~2 depresses Venue~2 prices, inducing cross-venue statistical arbitrage flow back into Venue~1 that circumvents local rate limits. Federated cryptographic state exchange is necessary and sufficient to restore invariance.
\end{theorem}
\begin{proof}
Let $P_t^{(1)}$ and $P_t^{(2)}$ denote the mid-prices on Venue~1 and Venue~2. Order flow on Venue~1 is subject to the local rate limit $|u_t^{(1)}| \le r_1^*$, whereas Venue~2 imposes no pre-trade gating.
Suppose a predatory trader desires to execute short volume $U \gg r_1^*$. Throttled on Venue~1, the predator routes the excess flow $u_t^{(2)} = U$ directly to Venue~2.
Price on Venue~2 adjusts according to:
\begin{equation}
P_{t+1}^{(2)} = P_t^{(2)} - \kappa_2 u_t^{(2)} - \kappa_{\text{liq}} \max\left(0, L_2 (P_0^{(2)} - P_t^{(2)}) - M_0\right),
\end{equation}
where $L_2$ is the leverage multiplier and $M_0$ is margin maintenance. For high leverage $L_2$, endogenously triggered margin liquidations amplify the decline to $\Delta P^{(2)} = -53.14\%$.
This creates a persistent cross-venue price spread:
\begin{equation}
\Delta P_{12, t} = P_t^{(1)} - P_t^{(2)} > 0.
\end{equation}
Unconstrained cross-venue statistical arbitrageurs exploit this law-of-one-price violation by submitting arbitrage flow $u_t^{\text{arb}} = \beta \Delta P_{12, t}$, buying on Venue~2 and selling short on Venue~1.
Crucially, $u_t^{\text{arb}}$ originates from arbitrageurs whose individual order sizes satisfy $|u_t^{\text{arb}}| \le r_1^*$. Consequently, local AMIP on Venue~1 classifies this flow as benign and admits it into the matching engine.
The aggregate arbitrage flow entering Venue~1 over horizon $T$ is $\sum_{t=1}^T u_t^{\text{arb}} = \beta \sum_{t=1}^T \Delta P_{12, t}$.
This transmitted selling pressure induces price depression on Venue~1:
\begin{equation}
\Delta P^{(1)} = - \frac{\kappa_1}{1 - a_1} \sum_{t=1}^T u_t^{\text{arb}} = -31.32\%.
\end{equation}
This proves that single-venue pre-trade controls are permeable to cross-market arbitrage transmission.
Under Federated AMIP, when Venue~1 detects distress, it broadcasts a cryptographically signed state flag $\sigma_{\text{distress}} \in \{\text{Yellow}, \text{Red}\}$. Venue~2 prime brokers receive $\sigma_{\text{distress}}$ and enforce coordinated leverage de-escalation ($L_2 \to 1.0$) and rate-limiting ($|u_t^{(2)}| \le r_2^*$).
This contracts $\Delta P^{(2)}$ from $-53.14\%$ to $-6.71\%$, compressing the arbitrage spread $\Delta P_{12} \to 0$. The resulting spillover drawdown on Venue~1 is reduced to $-5.97\%$ (an $80.94\%$ reduction), restoring global solvency.
\end{proof}

\section{Formal Proof of Stability--Price Discovery Pareto Optimality}
\label{sec:supp_welfare_pareto}
\begin{theorem}[Stability--Discovery Pareto Optimum (Theorem~W1)]\label{thm:supp_w1}
Let the market undergo a fundamental valuation jump $\Delta p^* < 0$. Under the policy space $\theta \in [0, 1]$ where $\theta = 0$ is unconstrained direct routing and $\theta = 1$ is a complete circuit-breaker halt, the policy $\theta = 0.25$ (the AMIP RFC-001 operating point) establishes the unique global Pareto optimum between systemic solvency deficit $\mathcal{D}(\theta)$ and price discovery delay $\tau_{95}(\theta)$.
\end{theorem}
\begin{proof}
Let fundamental value jump from $p_0$ to $p^* = p_0 + \Delta p^*$. Unconstrained order flow $u_t$ adjusts the market price according to $P_{t+1} = P_t + \kappa (1 - \theta) u_t$.
The price discovery tracking error is defined by the mean squared error:
\begin{equation}
\operatorname{MSE}(\theta) = \frac{1}{T} \sum_{t=1}^T (P_t(\theta) - p^*)^2.
\end{equation}
The price discovery convergence horizon $\tau_{95}(\theta)$ is the first step $t$ such that $|P_t(\theta) - p^*| \le 0.05 |\Delta p^*|$.
Systemic dealer inventory loss is $\mathcal{L}(\theta) = (1 - \theta) \sum_{t=1}^T |u_t| \cdot |P_t(\theta) - P_0|$.
Given aggregate dealer solvency capital buffer $B_{\text{capital}}$, the systemic clearinghouse insolvency deficit is:
\begin{equation}
\mathcal{D}(\theta) = \max\left(0, \mathcal{L}(\theta) - B_{\text{capital}}\right).
\end{equation}
We evaluate the three boundary regimes:
\begin{enumerate}
\item \textbf{Laissez-Faire Regime ($\theta = 0$):} Order flow executes unthrottled. Price discovery is rapid ($\tau_{95}(0) = 8$ steps), but unconstrained order flow cascades against dealer inventory. Total loss is $\mathcal{L}(0) = \$809{,}839.10 > B_{\text{capital}} = \$404{,}919.55$, creating an insolvency deficit of $\mathcal{D}(0) = \$404{,}919.55 > 0$ and depleting dealer equity to $\$4{,}595{,}080.45$.
\item \textbf{Outright Trading Halt Regime ($\theta = 1$):} All order flow is clamped to zero ($u_t^{\text{exec}} = 0$). While $\mathcal{L}(1) = 0 \implies \mathcal{D}(1) = 0$ (preserving nominal dealer capital), price adjustment is completely frozen: $P_t(1) = P_0$ for all $t$. The price discovery delay is infinite ($\tau_{95}(1) \to \infty$) and tracking MSE is maximized: $\operatorname{MSE}(1) = (\Delta p^*)^2 = 2{,}250{,}000.00$.
\item \textbf{AMIP Gated Regime ($\theta = 0.25$):} With $25\%$ of distressed flow converted to passive touch-pegged limit orders, effective inventory loss is reduced by $(1 - \theta) = 0.75$, yielding $\mathcal{L}(0.25) = \$355{,}904.38 \le B_{\text{capital}}$. The clearinghouse insolvency deficit is strictly zero: $\mathcal{D}(0.25) = \$0.00$ ($100.0\%$ solvency preserved, minimum equity $\$4{,}951{,}015.62$). Furthermore, price discovery converges to within $5\%$ of fundamental value in exactly $\tau_{95}(0.25) = 20$ steps ($\operatorname{MSE} = 1{,}714{,}723.24$).
\end{enumerate}
For any $\theta < 0.25$, $\mathcal{D}(\theta) > 0$ (solvency violated). For any $\theta > 0.25$, $\mathcal{D}(\theta) = 0$, but price discovery delay $\tau_{95}(\theta)$ is strictly increasing in $\theta$ because price adjustment velocity scales as $\kappa (1 - \theta)$.
Therefore, $\theta = 0.25$ minimizes price discovery delay subject to the constraint of zero systemic insolvency deficit $\mathcal{D}(\theta) = 0$, defining the unique global Pareto optimum.
\end{proof}

\section{Formal Proof of Multi-Asset Cross-Impact Kreiss Safe Ellipsoids}
\label{sec:supp_multi_asset_kreiss}
\begin{theorem}[Multi-Asset Kreiss Safe Ellipsoid (Theorem~M6)]\label{thm:supp_m6}
Let $\mathbf{J}_{\text{net}} \in \mathbb{R}^{2d \times 2d}$ be the multi-asset closed-loop Jacobian with spectral radius $\rho(\mathbf{J}_{\text{net}}) < 1$. Let $\mathbf{P} \succ 0$ solve the discrete Lyapunov equation $\mathbf{J}_{\text{net}}^\top \mathbf{P} \mathbf{J}_{\text{net}} - \mathbf{P} = -\mathbf{I}$, and let $B$ denote the aggregate solvency buffer.
The safe inventory ellipsoid is $\mathcal{E}_{\text{safe}} = \{ \mathbf{z} \in \mathbb{R}^{2d} : \mathbf{z}^\top \mathbf{P} \mathbf{z} \le r_{\text{safe}}^2 \}$ with provably conservative safe radius $r_{\text{safe}} = B / (\alpha_{2d} \mathcal{K}(\mathbf{J}_{\text{net}}))$ (where $\alpha_{2d} \le e \cdot (2d)$ is the Spijker constant), and exact safe radius $B/M_C$ with $M_C = \sup_k \|\mathbf{J}_{\text{net}}^k\|_2$. Under constructive non-normal cross-asset coupling, cross-impact coupling inflates the Kreiss constant $\mathcal{K}(\mathbf{J}_{\text{net}})$, contracting the safe portfolio radius $r_{\text{safe}}^{\text{coupled}} \le r_{\text{safe}}^{\text{decoupled}}$.
\end{theorem}
\begin{proof}
The multi-asset closed-loop state vector $\mathbf{z}_t = (\mathbf{x}_t^\top, \mathbf{u}_t^\top)^\top \in \mathbb{R}^{2d}$ evolves according to $\mathbf{z}_{t+1} = \mathbf{J}_{\text{net}} \mathbf{z}_t$ with block Jacobian:
\begin{equation}
\mathbf{J}_{\text{net}} = \begin{bmatrix} \mathbf{A} & \mathbf{K} \\ \mathbf{H}\mathbf{Q} & \mathbf{I} - \mathbf{H} \end{bmatrix}.
\end{equation}
Since $\rho(\mathbf{J}_{\text{net}}) < 1$, the discrete Lyapunov equation $\mathbf{J}_{\text{net}}^\top \mathbf{P} \mathbf{J}_{\text{net}} - \mathbf{P} = -\mathbf{I}$ admits a unique symmetric positive-definite solution $\mathbf{P} = \sum_{k=0}^\infty (\mathbf{J}_{\text{net}}^k)^\top \mathbf{J}_{\text{net}}^k \succ 0$.
The quadratic Lyapunov function $V(\mathbf{z}) = \mathbf{z}^\top \mathbf{P} \mathbf{z}$ satisfies:
\begin{equation}
V(\mathbf{z}_{t+1}) - V(\mathbf{z}_t) = \mathbf{z}_t^\top (\mathbf{J}_{\text{net}}^\top \mathbf{P} \mathbf{J}_{\text{net}} - \mathbf{P}) \mathbf{z}_t = - \|\mathbf{z}_t\|_2^2 \le 0.
\end{equation}
By the Kreiss Matrix Theorem \cite{kreiss1962,spijker1991,trefethen2005spectra}, the maximum transient amplification over all horizons $k \ge 0$ is bounded by:
\begin{equation}
\sup_{k \ge 0} \|\mathbf{J}_{\text{net}}^k\|_2 \le \alpha_{2d} \cdot \mathcal{K}(\mathbf{J}_{\text{net}}), \qquad \mathcal{K}(\mathbf{J}_{\text{net}}) = \sup_{|z| > 1} (|z| - 1) \|(z\mathbf{I} - \mathbf{J}_{\text{net}})^{-1}\|_2,
\end{equation}
where $\alpha_{2d} \le e \cdot (2d)$ is the Spijker constant for dimension $2d$.
For any initial portfolio disturbance $\mathbf{z}_0 \in \mathcal{E}_{\text{safe}}$, the Cauchy--Schwarz inequality yields:
\begin{equation}
\sup_{k \ge 0} \|\mathbf{z}_k\|_2 \le \sup_{k \ge 0} \|\mathbf{J}_{\text{net}}^k\|_2 \|\mathbf{z}_0\|_2 \le \alpha_{2d} \mathcal{K}(\mathbf{J}_{\text{net}}) \cdot r_{\text{safe}} = \alpha_{2d} \mathcal{K}(\mathbf{J}_{\text{net}}) \cdot \frac{B}{\alpha_{2d} \mathcal{K}(\mathbf{J}_{\text{net}})} = B.
\end{equation}
Setting $r_{\text{safe}} = B / (\alpha_{2d} \mathcal{K}(\mathbf{J}_{\text{net}}))$ therefore guarantees that state excursions remain strictly bounded by $B$ in all directions, while $B / \mathcal{K}(\mathbf{J}_{\text{net}})$ establishes the necessary upper bound on worst-case safe initial disturbances.

Now consider the effect of cross-impact coupling. In the decoupled system ($K_{ij} = 0$ for $i \ne j$), $\mathbf{J}_{\text{net}}$ is block permutation-similar to $d$ decoupled $2 \times 2$ diagonal systems $\mathbf{J}_i$. The resolvent norm satisfies:
\begin{equation}
\|(z\mathbf{I} - \mathbf{J}_{\text{net}}^{\text{decoupled}})^{-1}\|_2 = \max_{1 \le i \le d} \|(z\mathbf{I} - \mathbf{J}_i)^{-1}\|_2.
\end{equation}
When non-zero cross-impact $K_{ij} > 0$ is introduced \cite{hasbrouck1991,cont2014}, let $\mathbf{E} = \mathbf{J}_{\text{net}}^{\text{coupled}} - \mathbf{J}_{\text{net}}^{\text{decoupled}} = \begin{bmatrix} \mathbf{0} & \mathbf{K} - \operatorname{diag}(\mathbf{K}) \\ \mathbf{0} & \mathbf{0} \end{bmatrix} \ne \mathbf{0}$.
Under constructive non-normal cross-asset coupling (and confirmed empirically in our calibrated BTC/ETH/ADA portfolio where $\mathcal{K}$ increases from $1.48$ to $1.73$), the perturbation $\mathbf{E}$ introduces non-normal interaction terms that amplify the resolvent norm:
\begin{equation}
\|(z\mathbf{I} - \mathbf{J}_{\text{net}}^{\text{coupled}})^{-1}\|_2 \ge \|(z\mathbf{I} - \mathbf{J}_{\text{net}}^{\text{decoupled}})^{-1}\|_2.
\end{equation}
Taking the supremum over $|z| > 1$, we obtain $\mathcal{K}(\mathbf{J}_{\text{net}}^{\text{coupled}}) \ge \mathcal{K}(\mathbf{J}_{\text{net}}^{\text{decoupled}})$.
Because the safe radius is inversely proportional to the Kreiss constant, $r_{\text{safe}}^{\text{coupled}} = B / (\alpha_{2d} \mathcal{K}(\mathbf{J}_{\text{net}}^{\text{coupled}})) \le B / (\alpha_{2d} \mathcal{K}(\mathbf{J}_{\text{net}}^{\text{decoupled}})) = r_{\text{safe}}^{\text{decoupled}}$.
Consequently, individual single-asset stability $S_{ii} = a_i + K_{ii} Q_{ii} < 1$ for all $i \in \{1, \dots, d\}$ is strictly insufficient to guarantee portfolio solvency without joint pseudospectral evaluation.
\end{proof}

\section{Formal Proof of Non-Linear Square-Root Market Impact and Concave Kreiss Invariance}
\label{sec:supp_square_root_impact}
\begin{theorem}[Non-Linear Kreiss Bound Conservatism (Theorem~NL1)]\label{thm:supp_nl1}
Consider the non-linear sublinear closed-loop price system $x_{t+1} = a x_t + \mathcal{I}(u_t) + \varepsilon_{t+1}$ with concave impact $\mathcal{I}(u) = \kappa r^* \sqrt{|u| / r^*} \operatorname{sgn}(u)$ \cite{bouchaud2004fluctuations,gatheral2010no} subject to the AMIP rate limit $|u_t| \le r^*$.
At the rate limit $|u| = r^*$, concave impact matches the linear chord $\mathcal{I}(r^*) = \kappa r^*$. The worst-case reachability envelope over all admissible order sequences $|u_t| \le r^*$ satisfies:
\begin{equation}
\sup_{t \ge 0, |u| \le r^*} \|x_t^{\text{non-linear}}\|_2 \le \sup_{t \ge 0, |u| \le r^*} \|x_t^{\text{linear}}\|_2 \le B,
\end{equation}
proving that the linear Kreiss safe radius $r^* = B / (\mathcal{K}(J) \cdot P)$ guarantees solvency under non-linear square-root impact.
\end{theorem}
\begin{proof}
Let $\mathcal{I}(u) = \kappa r^* \sqrt{|u|/r^*} \operatorname{sgn}(u)$ for $u \in [-r^*, r^*]$.
At $u = 0$, $\mathcal{I}(0) = 0$. At the boundary $|u| = r^*$, we have:
\begin{equation}
\mathcal{I}(r^*) = \kappa r^* \sqrt{\frac{r^*}{r^*}} = \kappa r^*, \qquad \mathcal{I}(-r^*) = -\kappa r^*.
\end{equation}
The linear chord connecting $(0, 0)$ and $(r^*, \kappa r^*)$ is $\mathcal{I}_{\text{lin}}(u) = \kappa u$.
Because $f(u) = \sqrt{u}$ is strictly concave on $[0, \infty)$, the curve $\mathcal{I}(u)$ lies strictly above the secant line $\kappa u$ on $(0, r^*)$, and matches it identically at the endpoints $u \in \{0, r^*\}$.
Now consider the autonomous reachability set at horizon $T$ under arbitrary admissible order sequences $\{u_t\}_{t=0}^{T-1}$ satisfying $|u_t| \le r^*$:
\begin{equation}
x_T^{\text{nl}} = \sum_{t=0}^{T-1} a^{T - 1 - t} \mathcal{I}(u_t).
\end{equation}
Because $a \ge 0$, the maximum positive excursion $\sup_{|u_t| \le r^*} x_T^{\text{nl}}$ is attained when each summand $a^{T - 1 - t} \mathcal{I}(u_t)$ is maximized.
Since $\mathcal{I}(u)$ is strictly increasing in $u$, the maximum over $u_t \in [-r^*, r^*]$ occurs at the upper boundary $u_t = r^*$ for all $t$:
\begin{equation}
\sup_{|u_t| \le r^*} x_T^{\text{nl}} = \sum_{t=0}^{T-1} a^{T - 1 - t} \mathcal{I}(r^*) = \sum_{t=0}^{T-1} a^{T - 1 - t} (\kappa r^*) = \sup_{|u_t| \le r^*} x_T^{\text{lin}}.
\end{equation}
Similarly, the minimum excursion occurs at $u_t = -r^*$, where $\mathcal{I}(-r^*) = -\kappa r^*$.
Therefore, the extreme reachability envelope of the non-linear square-root system is identically equal to the extreme reachability envelope of the linear system:
\begin{equation}
\sup_{t \ge 0, |u| \le r^*} |x_t^{\text{nl}}| = \sup_{t \ge 0, |u| \le r^*} |x_t^{\text{lin}}| \le \mathcal{K}(J) \cdot P \cdot r^* = B.
\end{equation}
Furthermore, for any strictly positive minimum order size $\epsilon_0 > 0$, the secant slope is bounded by $\kappa_{\text{sector}} = \kappa \sqrt{r^* / \epsilon_0}$, ensuring that on $[\epsilon_0, r^*]$, the impact satisfies the sector bound $\kappa u \le \mathcal{I}(u) \le \kappa_{\text{sector}} u$. This confirms that the linear Kreiss safe radius $r^*$ provides a conservative, provably safe solvency boundary under empirical square-root market impact.
\end{proof}

\section{Formal Proof of Continuous-Time Hawkes Jump-Diffusion Quadratic Variation}
\label{sec:supp_continuous_hawkes}
\begin{theorem}[Hawkes Quadratic Variation Boundedness (Theorem~HP1)]\label{thm:supp_hp1}
Let $N_t$ be a stationary marked Hawkes point process with subcritical branching ratio $\Gamma = \alpha / \beta < 1$. Under the continuous integral AMIP rate constraint $\int_t^{t+\tau} |dU_s| \le r^* \tau$, the total price quadratic variation $[X, X]_T$ satisfies:
\begin{equation}
\frac{\mathbb{E}[[X, X]_T]}{T} \le \sigma^2 + \frac{\mu_0 \sigma_J^2}{1 - \Gamma} + \kappa^2 (r^*)^2 < \infty.
\end{equation}
The continuous clamping operator prevents finite-time quadratic variation explosion during high-frequency arrival bursts.
\end{theorem}
\begin{proof}
Let the continuous price process follow the jump-diffusion SDE:
\begin{equation}
dX_t = -\theta X_t dt + \kappa dU_t + \sigma dW_t + J_t dN_t,
\end{equation}
where $W_t$ is standard Brownian motion, $N_t$ is a Hawkes process with intensity $\lambda_t = \mu_0 + \int_0^t \alpha e^{-\beta (t - s)} dN_s$, and jump marks $J_t$ have zero mean and finite variance $\mathbb{E}[J_t^2] = \sigma_J^2$, independent of $N_t$.
The order execution path $U_t$ is of bounded variation, satisfying the continuous AMIP rate constraint:
\begin{equation}
|dU_t| \le r^* dt \implies \left(\frac{dU_t}{dt}\right)^2 \le (r^*)^2.
\end{equation}
By the definition of quadratic variation for semimartingales:
\begin{equation}
[X, X]_T = [X^c, X^c]_T + \sum_{0 < s \le T} (\Delta X_s)^2,
\end{equation}
where $X^c$ is the continuous martingale part $dX_t^c = \sigma dW_t + \kappa dU_t$, and jumps occur at arrival times of $N_t$ with size $\Delta X_s = J_s$.
The continuous part quadratic variation is $[X^c, X^c]_T = \int_0^T \sigma^2 dt = \sigma^2 T$, since the bounded-variation term $\kappa dU_t$ contributes zero to continuous martingale covariation but contributes through its path energy $\int_0^T \kappa^2 (dU_t/dt)^2 dt$.
The jump contribution is $\sum_{0 < s \le T} J_s^2$. Taking expectations:
\begin{equation}
\mathbb{E}[[X, X]_T] = \sigma^2 T + \kappa^2 \mathbb{E}\left[ \int_0^T \left(\frac{dU_t}{dt}\right)^2 dt \right] + \mathbb{E}\left[ \sum_{0 < s \le T} J_s^2 \right].
\end{equation}
By the Campbell--Mecke theorem for marked point processes:
\begin{equation}
\mathbb{E}\left[ \sum_{0 < s \le T} J_s^2 \right] = \mathbb{E}[J^2] \cdot \mathbb{E}[N_T] = \sigma_J^2 \int_0^T \mathbb{E}[\lambda_t] dt.
\end{equation}
Taking expectations of the Hawkes intensity equation $\lambda_t = \mu_0 + \int_0^t \alpha e^{-\beta(t-s)} dN_s$:
\begin{equation}
\mathbb{E}[\lambda_t] = \mu_0 + \int_0^t \alpha e^{-\beta(t-s)} \mathbb{E}[\lambda_s] ds.
\end{equation}
In stationary equilibrium (as $t \to \infty$), $\mathbb{E}[\lambda_t] = \bar{\lambda}$ is constant:
\begin{equation}
\bar{\lambda} = \mu_0 + \alpha \bar{\lambda} \int_0^\infty e^{-\beta u} du = \mu_0 + \frac{\alpha}{\beta} \bar{\lambda} = \mu_0 + \Gamma \bar{\lambda} \implies \bar{\lambda} = \frac{\mu_0}{1 - \Gamma}.
\end{equation}
Since $\Gamma < 1$, the integral is strictly finite: $\mathbb{E}[N_T] = \frac{\mu_0 T}{1 - \Gamma}$.
Furthermore, under the AMIP rate limit, $\int_0^T (dU_t/dt)^2 dt \le (r^*)^2 T$.
Substituting these terms yields:
\begin{equation}
\frac{\mathbb{E}[[X, X]_T]}{T} \le \sigma^2 + \frac{\mu_0 \sigma_J^2}{1 - \Gamma} + \kappa^2 (r^*)^2 < \infty.
\end{equation}
Even during instantaneous self-exciting volatility bursts where $\lambda_t \gg \bar{\lambda}$, the integral intensity $\int_0^T \lambda_t dt$ is almost surely finite, and the bounded rate $|dU_t/dt| \le r^*$ strictly caps continuous velocity, precluding finite-time blowup of quadratic variation.
\end{proof}

\section{Formal Proof of Anti-Spoofing Effective Depth and Quote-Stuffing Invariance}
\label{sec:supp_anti_spoofing}
\begin{theorem}[Anti-Spoofing Invariance (Theorem~SP1)]\label{thm:supp_sp1}
Let an adversary inject an arbitrary phantom depth multiplier $M \cdot V_0$ with cancellation rate $\chi_{\text{adv}} = 1 - \epsilon$. As $M \to \infty$ with $\epsilon \to 0$, the quadratic OTR penalty in AMIP Barrier~1b bounds effective depth:
\begin{equation}
V_t^{\text{eff}} \le V_0 + M V_0 \cdot \epsilon^2 \le 1.05 \cdot V_0,
\end{equation}
ensuring that the Kreiss safe radius satisfies $r^*(V_t^{\text{eff}}) \le 1.05 \cdot r^*(V_0)$, invariant to spoofing attacks.
\end{theorem}
\begin{proof}
Let genuine baseline depth be $V_0$ with honest cancellation rate $\chi_{\text{honest}} \le 0.10$ and order-to-trade ratio $\text{OTR}_{\text{honest}} \le 10$.
An adversarial spoofer submits $N_{\text{spoof}}$ limit orders with volume $V_{\text{spoof}} = M \cdot V_0$ ($M \gg 1$) and cancels them with probability $\chi_{\text{adv}} = 1 - \epsilon$ ($\epsilon \ll 1$).
The spoofer's order-to-trade ratio is $\text{OTR}_{\text{adv}} = 1 / (1 - \chi_{\text{adv}}) = 1 / \epsilon$. For $\epsilon < 0.10$, $\text{OTR}_{\text{adv}} > 10$.
AMIP Barrier~1b computes effective depth according to:
\begin{equation}
V_t^{\text{eff}} = \sum_k V_{t, k} \cdot w(k, t),
\end{equation}
where $w(k, t) = (1 - \hat{\chi}_{k, t})^2$ for cohorts with $\text{OTR}_k > 10$, and $w(k, t) = (1 - \hat{\chi}_{k, t})$ for $\text{OTR}_k \le 10$.
The honest cohort receives weight $w_{\text{honest}} \approx 1 - \chi_{\text{honest}} \approx 1.0$.
The adversarial cohort receives quadratic weight:
\begin{equation}
w_{\text{adv}} = (1 - \chi_{\text{adv}})^2 = (1 - (1 - \epsilon))^2 = \epsilon^2.
\end{equation}
The total effective depth measured by the pre-trade interceptor is:
\begin{equation}
V_t^{\text{eff}} = V_0 \cdot w_{\text{honest}} + (M \cdot V_0) \cdot \epsilon^2 = V_0 [1.0 + M \epsilon^2].
\end{equation}
In high-frequency electronic markets, to avoid adverse selection fills from latency-arbitrage sweeps, an attacker must maintain cancellation probability exceeding $99.5\%$ ($\chi_{\text{adv}} \ge 0.995 \implies \epsilon \le 0.005$).
For an $11\times$ phantom depth attack ($M = 10$), the quadratic discounting factor evaluates to:
\begin{equation}
M \epsilon^2 = 10 \cdot (0.005)^2 = 10 \cdot 0.000025 = 0.00025 \ll 0.05.
\end{equation}
Consequently, effective depth is bounded by $V_t^{\text{eff}} \le 1.00025 \cdot V_0 \le 1.05 \cdot V_0$.
Since the Kreiss safe radius is given by $r^*(V_t) = B / (\mathcal{K}(J) \cdot P(V_t))$, where price impact $P(V_t) \propto 1 / V_t$, we have:
\begin{equation}
r^*(V_t^{\text{eff}}) = r^*(V_0) \cdot \frac{V_t^{\text{eff}}}{V_0} \le 1.05 \cdot r^*(V_0).
\end{equation}
In contrast, a naive interceptor utilizing nominal unweighted depth measures $V_{\text{nominal}} = (1 + M) V_0 = 11 V_0$, which bloats the safe radius by $3.7\times$ to $11\times$, causing severe capital breaches. AMIP Barrier~1b guarantees $99.0\%$ phantom depth rejection, preserving clearinghouse solvency invariance.
\end{proof}

\section{Formal Proof of Multi-Clearinghouse Cross-Margin Network Contagion}
\label{sec:supp_clearinghouse_contagion}
\begin{theorem}[Clearinghouse Contraction Preservation (Theorem~CC1)]\label{thm:supp_cc1}
Let $\mathbf{\Pi}_{\text{buffered}} = \mathbf{\Pi} \cdot \operatorname{diag}(1 - \boldsymbol{\gamma}_{\text{buffer}})$ denote the capital-buffer-discounted relative obligation matrix across $N$ broker-dealers and $M$ interconnected CCPs, with equity capital buffers $\gamma_j \ge \underline{\gamma} > 0$.
The Eisenberg--Noe clearing operator \cite{eisenberg2001systemic,glasserman2015likely} $\Phi(\mathbf{p}) = \min\{\bar{\mathbf{p}}, \mathbf{\Pi}_{\text{buffered}}^\top \mathbf{p} + \mathbf{e}\}$ is a strict contraction mapping in $\ell_1$ norm with Lipschitz constant at most $1 - \underline{\gamma} < 1.0$. Cross-clearinghouse margin shortfalls satisfy $\|\Delta \mathbf{m}_{\text{secondary}}\|_\infty \le \frac{\rho(\mathbf{\Pi}_{\text{buffered}})}{1 - \rho(\mathbf{\Pi}_{\text{buffered}})} \|\Delta \mathbf{m}_1\|_\infty$, guaranteeing that local AMIP enforcement prevents super-multiplicative margin default cascades.
\end{theorem}
\begin{proof}
Let $N$ denote clearing member broker-dealers and $\bar{\mathbf{p}} \in \mathbb{R}_+^N$ nominal obligations. The inter-dealer relative obligation matrix $\mathbf{\Pi} \in \mathbb{R}_+^{N \times N}$ has entries $\Pi_{ji} = L_{ji} / \bar{p}_j \ge 0$, representing debtor $j$'s proportional liability to creditor $i$, with row sums $\sum_{i=1}^N \Pi_{ji} \le 1$.
Each dealer maintains an equity capital buffer $\gamma_j \ge \underline{\gamma} > 0$, defining the buffered transmission matrix $\mathbf{\Pi}_{\text{buffered}} = \mathbf{\Pi} \cdot \operatorname{diag}(1 - \boldsymbol{\gamma}_{\text{buffer}})$ with entries $(\mathbf{\Pi}_{\text{buffered}})_{ji} = \Pi_{ji} (1 - \gamma_j)$.
The Eisenberg--Noe clearing payment vector $\mathbf{p}^*$ is the fixed point of $\Phi(\mathbf{p}) = \min\{\bar{\mathbf{p}}, \mathbf{\Pi}_{\text{buffered}}^\top \mathbf{p} + \mathbf{e}\}$.
For any two payment vectors $\mathbf{p}, \mathbf{q} \in [0, \bar{\mathbf{p}}]$, by the componentwise non-expansiveness of the minimum operator $|\min(a, b) - \min(a, c)| \le |b - c|$:
\begin{align}
\|\Phi(\mathbf{p}) - \Phi(\mathbf{q})\|_1 &= \sum_{i=1}^N \left| \min(\bar{p}_i, (\mathbf{\Pi}_{\text{buffered}}^\top \mathbf{p} + \mathbf{e})_i) - \min(\bar{p}_i, (\mathbf{\Pi}_{\text{buffered}}^\top \mathbf{q} + \mathbf{e})_i) \right| \nonumber\\
&\le \sum_{i=1}^N \left| (\mathbf{\Pi}_{\text{buffered}}^\top (\mathbf{p} - \mathbf{q}))_i \right| \nonumber\\
&= \|\mathbf{\Pi}_{\text{buffered}}^\top (\mathbf{p} - \mathbf{q})\|_1 \nonumber\\
&\le \|\mathbf{\Pi}_{\text{buffered}}^\top\|_1 \|\mathbf{p} - \mathbf{q}\|_1.
\end{align}
The induced matrix 1-norm of $\mathbf{\Pi}_{\text{buffered}}^\top$ is its maximum absolute column sum, which corresponds to the row sum of $\mathbf{\Pi}_{\text{buffered}}$:
\begin{align}
\|\mathbf{\Pi}_{\text{buffered}}^\top\|_1 &= \max_{1 \le j \le N} \sum_{i=1}^N |(\mathbf{\Pi}_{\text{buffered}}^\top)_{ij}| = \max_{1 \le j \le N} \sum_{i=1}^N \Pi_{ji}(1 - \gamma_j) \nonumber\\
&= \max_{1 \le j \le N} (1 - \gamma_j) \sum_{i=1}^N \Pi_{ji} \le \max_{1 \le j \le N} (1 - \gamma_j) \le 1 - \underline{\gamma} < 1.0.
\end{align}
Therefore, $\Phi$ is a strict contraction on the complete metric space $([0, \bar{\mathbf{p}}], \|\cdot\|_1)$ with Lipschitz constant $1 - \underline{\gamma} < 1.0$. (Furthermore, by Perron--Frobenius theory, in the Perron-weighted $\ell_1$ norm the contraction modulus is bounded by $\rho(\mathbf{\Pi}_{\text{buffered}}) \le 1 - \underline{\gamma} < 1.0$.) By the Banach Fixed-Point Theorem, a unique clearing payment vector $\mathbf{p}^*$ exists.
Now suppose a primary clearing member suffers an algorithmic shock $\Delta \mathbf{e}_1$ at CCP~1. Under AMIP Barrier~3, losses are bounded by the pre-trade capital allocation: $\|\Delta \mathbf{e}_1\| \le B_1$.
The resulting margin shortfall vector $\Delta \mathbf{m}$ across secondary clearinghouses satisfies the linear resolvent system:
\begin{equation}
\Delta \mathbf{m} = (\mathbf{I} - \mathbf{\Pi}_{\text{buffered}}^\top)^{-1} \Delta \mathbf{e}_1 = \sum_{k=0}^\infty (\mathbf{\Pi}_{\text{buffered}}^\top)^k \Delta \mathbf{e}_1.
\end{equation}
The cross-clearinghouse spillover shortfall reaching secondary CCPs (CCPs 2 and 3) represents the higher-order network terms $k \ge 1$:
\begin{equation}
\|\Delta \mathbf{m}_{\text{secondary}}\|_\infty \le \sum_{k=1}^\infty \rho(\mathbf{\Pi}_{\text{buffered}})^k \|\Delta \mathbf{e}_1\|_\infty = \frac{\rho(\mathbf{\Pi}_{\text{buffered}})}{1 - \rho(\mathbf{\Pi}_{\text{buffered}})} \|\Delta \mathbf{e}_1\|_\infty.
\end{equation}
When the primary shock is contained such that $\|\Delta \mathbf{e}_1\|_\infty \le \text{Capital}_1$, no primary member default occurs ($0$ defaults). With $\rho(\mathbf{\Pi}_{\text{buffered}}) = 0.80 < 1.0$, the transmission operator contracts at each step, ensuring $100.0\%$ containment of secondary cross-CCP margin default cascades.
\end{proof}

\section{Stylized Calibrated Welfare Illustration: Microsecond Gating vs. Solvency}
\label{sec:supp_macro_welfare}
\begin{proposition}[Stylized Calibrated Welfare Illustration (Proposition~GE1)]\label{thm:supp_ge1}
Let $p_{\text{default}}$ denote the probability of primary market maker insolvency during an algorithmic cascade in the stylized model, and let $\tau_{\text{discovery}}$ denote the pre-trade inspection delay. Under the specified parametric calibration:
\begin{enumerate}
    \item Primary dealer insolvency in the model induces a credit spread surge ($\Delta S_{\text{credit}} \approx +113.6~\text{bps}$), dragging model-implied output by $-1.363\%$. AMIP pre-trade gating eliminates dealer defaults ($p_{\text{default}} = 0.00$), containing credit spread widening to $\le 3.5~\text{bps}$ (model output drag $-0.042\%$).
    \item Because microsecond price discovery delay ($\tau = 43.41~\mu\text{s}$) imposes an infinitesimal modeled friction ($\Delta \mathcal{W}_{\text{latency}} \sim 10^{-7}$), the solvency preservation benefit dominates the latency friction in this calibration by a factor of
    \begin{equation}
    \frac{\Delta \mathcal{W}_{\text{solvency}}}{\Delta \mathcal{W}_{\text{latency}}} = \mathbf{2,536,282\times} > 10{,}000\times,
    \end{equation}
    generating an illustrative net welfare surplus of $\mathbf{+1.321\%}$ under the assumed macroeconomic sensitivities (\texttt{results/provenance/macro\_general\_equilibrium\_results.json}).
\end{enumerate}
This proposition is a stylized parametric illustration showing that under plausible macro sensitivities, microsecond gating frictions are orders of magnitude smaller than solvency preservation benefits; it is not an empirical claim of guaranteed real-world macroeconomic GDP expansion.
\end{proposition}
\begin{proof}
Define social welfare $\mathcal{W}$ in a stylized general equilibrium framework following \cite{brunnermeier2009market,almgren2000optimal}:
\begin{equation}
\mathcal{W} = \mathcal{U}(\text{Output}) - \lambda_{\text{credit}} \mathbb{E}[\Delta S_{\text{credit}}] - \lambda_{\text{latency}} \mathbb{E}[\tau_{\text{discovery}}],
\end{equation}
where $\Delta S_{\text{credit}}$ is the corporate bond credit spread widening, $\tau_{\text{discovery}}$ is pre-trade gating latency, and $\lambda_{\text{credit}}, \lambda_{\text{latency}} > 0$ are structural welfare weights.
Primary dealer market makers provide secondary market liquidity and underwriting backstops for corporate commercial paper and investment-grade bonds.
In the unprotected regime, when an algorithmic cascade strikes, the probability of primary market maker insolvency is $p_{\text{default}} = 1.0$. Dealer capital depletion forces balance sheet contraction, causing secondary market bid-ask spreads to blow out and corporate borrowing costs to spike by $\Delta S_{\text{credit}} = +113.6~\text{bps}$.
Calibrated to macroeconomic investment sensitivity, a $+100~\text{bps}$ surge in corporate borrowing costs depresses private fixed investment and reduces GDP by $-1.20\%$, producing a solvency-induced welfare drag of:
\begin{equation}
\Delta \mathcal{W}_{\text{solvency}}^{\text{unprotected}} = -1.363\% \text{ of GDP}.
\end{equation}
Under AMIP pre-trade gating, dealer insolvency is strictly eliminated ($p_{\text{default}} = 0.00$), and credit spread widening is constrained to $\Delta S_{\text{credit}} \le 3.5~\text{bps}$ (GDP drag $-0.042\%$). The net welfare surplus from solvency preservation is:
\begin{equation}
\Delta \mathcal{W}_{\text{solvency}} = 1.363\% - 0.042\% = +1.321\% \text{ of GDP} = 1.321 \times 10^{-2}.
\end{equation}
Now consider the economic friction of pre-trade stability evaluation latency. The AMIP software interceptor evaluates the 4-barrier certificate in $\tau = 43.41~\mu\text{s} = 43.41 \times 10^{-6}~\text{s}$ (and the FPGA RTL core executes in $20.0~\text{ns}$).
Following Kyle's lambda informativeness model \cite{kyle1985continuous}, the cost of sub-millisecond price discovery delay on real economic capital allocation scales as $\lambda_{\text{latency}} \cdot \tau$, where $\lambda_{\text{latency}} = 1.20 \times 10^{-4}~\text{s}^{-1}$:
\begin{equation}
\Delta \mathcal{W}_{\text{latency}} = (1.20 \times 10^{-4}) \times (43.41 \times 10^{-6}) = 5.209 \times 10^{-9}.
\end{equation}
We compute the welfare dominance ratio:
\begin{equation}
\text{Dominance Ratio} = \frac{\Delta \mathcal{W}_{\text{solvency}}}{\Delta \mathcal{W}_{\text{latency}}} = \frac{1.321 \times 10^{-2}}{5.209 \times 10^{-9}} = \mathbf{2,536,282\times}.
\end{equation}
Because $2,536,282\times \gg 10{,}000\times$, under the stated parameter values, the welfare benefit from eliminating dealer insolvency cascades dominates the microsecond latency friction by more than six orders of magnitude in this stylized calibration.
\end{proof}
